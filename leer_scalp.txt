#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""okx_scalp_bot.py - Bot de SCALPING para DEMO OKX (XPERPs)

Estrategia (Alex Ruiz / Trading Lab, adaptada a cripto 24/7):
  1H  DIRECCION : test de resistencia/soporte 24h + vela de giro + precio
                  del lado correcto de la EMA200(1H) [~ EMA50 de 4H]
  5M  SETUP     : impulso con ruptura de minimos/maximos anteriores +
                  pullback hasta la zona Fibonacci 0.382-0.75 del impulso
  1M  GATILLO   : cierre cruzando la EMA50(1M) + ruptura de la zona de
                  3 velas + volumen > 1.5x la media de 20
  SL en Fib 0.75 | TP en el extremo del impulso (R:R minimo 1.5)
  Trailing (cb 3%) al superar +1.5R, como FIXTS

Seguridad: riesgo fijo $5/trade, max 3 trades/dia, 1 posicion a la vez,
max 2 perdidas por moneda al dia, cooldown 15 min por moneda.
Kill switch:  touch /root/.okx_scalp_pausado   (gestiona abiertas, no abre)
Estado:       python3 /root/okx_scalp_bot.py --estado
"""
import json

# --------------------------- configuracion ---------------------------
MONEDAS = ["BTC", "ETH", "SOL", "XRP", "DOGE"]
RIESGO_USD = 5.0
MAX_TRADES_DIA = 3
MAX_PERDIDAS_MONEDA = 2
COOLDOWN_SEG = 15 * 60
PAUSA_TRAS_CIERRE = 300     # tras cerrar un trade, 5 min antes de abrir otro
ZONA_1H_PCT = 0.015          # cercania al extremo de 24h que cuenta como test
MIN_IMPULSO_PCT = 0.006      # tamano minimo del impulso en 5M (0.6%)
UMBRAL_VOL = 1.5             # volumen del gatillo vs media de 20
MIN_RR = 1.5                 # riesgo:beneficio minimo
MIN_DIST_SL = 0.0012         # distancia minima al SL (0.12% del precio)
MAX_NOTIONAL = 2000.0        # tope de exposicion por trade
TRAILING_DESDE_R = 1.5
TRAILING_CB = 0.03
CICLO_SEG = 60
HEARTBEAT_CADA = 15          # linea de estado cada N ciclos

ARCH_MANAGED = "/root/okx_scalp_managed.json"
ARCH_JOURNAL = "/root/okx_scalp_journal.json"
ARCH_PAUSA = "/root/.okx_scalp_pausado"

COOLDOWNS = {}
ULTIMO_CIERRE_TS = 0.0
TG_TOKEN = ""
TG_CHAT = ""


def tg_send(texto):
    """Envia a Telegram (mismo sistema que salud_bots). Nunca rompe el bot."""
    if not TG_TOKEN or not TG_CHAT:
        return False
    try:
        data = urllib.parse.urlencode({"chat_id": TG_CHAT, "text": texto}).encode()
        req = urllib.request.Request(
            "https://api.telegram.org/bot" + TG_TOKEN + "/sendMessage", data=data)
        urllib.request.urlopen(req, timeout=20)
        return True
    except Exception:
        return False


def avisar(texto):
    """Log + aviso por Telegram de lo importante (aperturas, cierres, trailing)."""
    registrar(texto)
    tg_send(texto)


# --------------------------- utilidades ---------------------------
def registrar(txt):
    print(time.strftime("[%Y-%m-%d %H:%M:%S] ") + txt, flush=True)


def cargar_json(ruta, defecto):
    try:
        return json.load(open(ruta))
    except Exception:
        return defecto


def guardar_json(ruta, datos):
    tmp = ruta + ".tmp"
    json.dump(datos, open(tmp, "w"), indent=1)
    os.replace(tmp, ruta)


def ema(vals, periodo):
    if len(vals) < periodo:
        return None
    k = 2.0 / (periodo + 1)
    e = sum(vals[:periodo]) / float(periodo)
    for v in vals[periodo:]:
        e = v * k + e * (1 - k)
    return e


def velas_cerradas(client, inst, barra, n):
    """Ultimas n velas CERRADAS (la API incluye la vela en curso: se descarta)."""
    todas = client.velas(inst, barra, n + 1) or []
    cerradas = todas[:-1] if len(todas) > n else todas
    return cerradas[-n:]


def hoy0h():
    t = time.gmtime()
    return time.mktime((t.tm_year, t.tm_mon, t.tm_mday, 0, 0, 0, 0, 0, 0))


# --------------------------- analisis 1H ---------------------------
def permiso_1h(v):
    """Direccion de las proximas 2-3 velas: 'SHORT', 'LONG' o None."""
    if len(v) < 210:
        return None
    cierres = [x["close"] for x in v]
    e200 = ema(cierres[-250:], 200)
    if e200 is None:
        return None
    ult = v[-1]
    max24 = max(x["high"] for x in v[-24:])
    min24 = min(x["low"] for x in v[-24:])
    toco_res = any(x["high"] >= max24 * (1 - ZONA_1H_PCT) for x in v[-3:])
    toco_sop = any(x["low"] <= min24 * (1 + ZONA_1H_PCT) for x in v[-3:])
    giro_baj = ult["close"] < ult["open"]
    giro_alc = ult["close"] > ult["open"]
    if toco_res and giro_baj and ult["close"] < e200:
        return "SHORT"
    if toco_sop and giro_alc and ult["close"] > e200:
        return "LONG"
    return None


# --------------------------- analisis 5M ---------------------------
def setup_5m(v, direccion):
    """Impulso + pullback a Fibonacci. Devuelve dict con sl/objetivo o None."""
    if len(v) < 40:
        return None
    w = v[-60:]
    if direccion == "SHORT":
        ih = max(range(len(w)), key=lambda i: w[i]["high"])
        if ih > len(w) - 4:
            return None
        il = min(range(ih, len(w)), key=lambda i: w[i]["low"])
        if il - ih < 3:
            return None
        h, l = w[ih]["high"], w[il]["low"]
        if ih == 0 or l >= min(x["low"] for x in w[:ih]):
            return None                       # no rompio minimos anteriores
        rango = h - l
        if rango / h < MIN_IMPULSO_PCT:
            return None
        tras_l = len(w) - 1 - il
        if not (3 <= tras_l <= 30):
            return None
        fib = lambda lvl: l + lvl * rango
        zona_lo, zona_hi = fib(0.382), fib(0.75)
        px = w[-1]["close"]
        if not (zona_lo <= px <= zona_hi):
            return None
        return {"extremo": h, "objetivo": l, "fib382": zona_lo, "fib75": zona_hi,
                "sl": fib(0.75), "velas_tras_L": tras_l}
    # LONG (espejo): impulso alcista con ruptura de maximos, pullback a la baja
    il = min(range(len(w)), key=lambda i: w[i]["low"])
    if il > len(w) - 4:
        return None
    ih = max(range(il, len(w)), key=lambda i: w[i]["high"])
    if ih - il < 3:
        return None
    l, h = w[il]["low"], w[ih]["high"]
    if il == 0 or h <= max(x["high"] for x in w[:il]):
        return None
    rango = h - l
    if rango / l < MIN_IMPULSO_PCT:
        return None
    tras_h = len(w) - 1 - ih
    if not (3 <= tras_h <= 30):
        return None
    fib = lambda lvl: h - lvl * rango
    zona_lo, zona_hi = fib(0.75), fib(0.382)
    px = w[-1]["close"]
    if not (zona_lo <= px <= zona_hi):
        return None
    return {"extremo": l, "objetivo": h, "fib382": zona_hi, "fib75": zona_lo,
            "sl": fib(0.75), "velas_tras_L": tras_h}


# --------------------------- analisis 1M ---------------------------
def gatillo_1m(v, direccion):
    """Ruptura de EMA50(1M) + zona de 3 velas + volumen alto, en vela cerrada."""
    if len(v) < 55:
        return False
    cierres = [x["close"] for x in v]
    e50 = ema(cierres, 50)
    if e50 is None:
        return False
    ult = v[-1]
    prev3 = v[-4:-1]
    vols = [x["vol"] for x in v[-21:-1]]
    vol_med = sum(vols) / len(vols) if vols else 0.0
    vol_ok = vol_med > 0 and ult["vol"] > UMBRAL_VOL * vol_med
    if direccion == "SHORT":
        return (ult["close"] < e50 and ult["close"] < ult["open"]
                and ult["low"] < min(x["low"] for x in prev3) and vol_ok)
    return (ult["close"] > e50 and ult["close"] > ult["open"]
            and ult["high"] > max(x["high"] for x in prev3) and vol_ok)


# --------------------------- journal ---------------------------
def journal():
    return cargar_json(ARCH_JOURNAL, [])


def trades_hoy():
    h0 = hoy0h()
    return [j for j in journal() if j.get("abierta_ts", 0) >= h0]


def perdidas_hoy(base):
    h0 = hoy0h()
    return len([j for j in journal()
                if j.get("moneda") == base and j.get("estado") == "cerrada"
                and j.get("cerrada_ts", 0) >= h0 and (j.get("pnl") or 0) < 0])


def anotar_apertura(reg):
    j = journal()
    j.append(reg)
    guardar_json(ARCH_JOURNAL, j)


def anotar_cierre(base, pnl, motivo):
    j = journal()
    for r in reversed(j):
        if r.get("moneda") == base and r.get("estado") == "abierta":
            r["estado"] = "cerrada"
            r["pnl"] = pnl
            r["motivo"] = motivo
            r["cerrada_ts"] = time.time()
            break
    guardar_json(ARCH_JOURNAL, j)


# --------------------------- abrir ---------------------------
def abrir(client, base, inst, direccion, s):
    entry = float(client.ticker(inst).get("last", 0) or 0)
    if entry <= 0:
        return None
    sl, tp = s["sl"], s["objetivo"]
    if direccion == "SHORT":
        dist, beneficio = sl - entry, entry - tp
    else:
        dist, beneficio = entry - sl, tp - entry
    if dist <= 0 or beneficio <= 0:
        registrar("%s: señal descartada (SL/TP del lado equivocado)" % base)
        return None
    if dist / entry < MIN_DIST_SL:
        registrar("%s: señal descartada (SL demasiado cerca: %.3f%%)" %
                  (base, dist / entry * 100))
        return None
    if beneficio / dist < MIN_RR:
        registrar("%s: señal descartada (R:R %.2f < %.2f)" %
                  (base, beneficio / dist, MIN_RR))
        return None
    info = client.info_instr(inst) or {}
    ctval = float(info.get("ctVal", 0) or 0)
    if ctval <= 0:
        registrar("%s: sin ctVal, me lo salto" % base)
        return None
    sz = int(RIESGO_USD / (dist * ctval))
    if sz * ctval * entry > MAX_NOTIONAL:
        sz = int(MAX_NOTIONAL / (ctval * entry))
    if sz < 1:
        registrar("%s: señal descartada (SL lejos: 1 contrato > riesgo $%s)" %
                  (base, RIESGO_USD))
        return None
    lado = "sell" if direccion == "SHORT" else "buy"
    cierre = "buy" if direccion == "SHORT" else "sell"
    r = client.orden_mercado(inst, lado, sz)
    if not r or str(r.get("code")) != "0":
        registrar("%s: FALLO orden mercado: %s" % (base, str(r)[:100]))
        return None
    sl_id = client.orden_algo_sl(inst, cierre, round(sl, 8), sz)
    tp_id = client.orden_algo_tp(inst, cierre, round(tp, 8), sz)
    reg = {"moneda": base, "direccion": direccion, "inst": inst,
           "entry": entry, "sl": sl, "tp": tp, "sz": sz, "ctVal": ctval,
           "rr": round(beneficio / dist, 2), "sl_id": sl_id, "tp_id": tp_id,
           "trail_id": None, "estado": "abierta", "abierta_ts": time.time()}
    guardar_json(ARCH_MANAGED, {base: reg})
    anotar_apertura(reg)
    avisar("🚀 SCALP %s %s abierta\n%d ct @ %.6g\nSL %.6g | TP %.6g\nR:R %.2f | riesgo $%s"
           % (base, direccion, sz, entry, sl, tp, reg["rr"], RIESGO_USD))
    COOLDOWNS[base] = time.time()
    return reg


# --------------------------- gestion ---------------------------
def gestionar(client):
    mg = cargar_json(ARCH_MANAGED, {})
    if not mg:
        return None
    base, m = list(mg.items())[0]
    inst = m.get("inst") or client.inst_id(base)
    pos = next((p for p in (client.posiciones() or [])
                if p.get("instId") == inst and float(p.get("pos", 0) or 0) != 0),
               None)
    if pos is None:
        # cerrada (SL, TP o trailing). Buscar P&L, limpiar algos, journal.
        pnl = None
        try:
            hist = client.posiciones_cerradas(inst, 5) or []
            if hist and float(hist[0].get("uTime", 0) or 0) > time.time() * 1000 - 3600000:
                pnl = float(hist[0].get("pnl", 0) or 0)
        except Exception:
            pass
        try:
            client.cancelar_todas_algo(inst)
        except Exception:
            pass
        anotar_cierre(base, pnl, "SL/TP/trailing")
        guardar_json(ARCH_MANAGED, {})
        COOLDOWNS[base] = time.time()
        global ULTIMO_CIERRE_TS
        ULTIMO_CIERRE_TS = time.time()
        txt_pnl = ("%.2f USD" % pnl) if pnl is not None else "sin dato"
        txt_r = (" (%+.1fR)" % (pnl / RIESGO_USD)) if pnl is not None else ""
        h0 = hoy0h()
        cerradas_hoy = [j for j in journal() if j.get("estado") == "cerrada"
                        and j.get("cerrada_ts", 0) >= h0]
        total_hoy = sum(j.get("pnl") or 0 for j in cerradas_hoy)
        resumen = ("\nHoy: %d trades, %+.2f USD" % (len(cerradas_hoy), total_hoy)
                   if cerradas_hoy else "")
        avisar("🏁 SCALP %s cerrada\nP&L: %s%s%s\n(cooldown %d min)"
               % (base, txt_pnl, txt_r, resumen, COOLDOWN_SEG // 60))
        return None
    upl = float(pos.get("upl", 0) or 0)
    entry_vivo = float(pos.get("avgPx", 0) or 0) or m.get("entry", 0)
    r_dist = abs(entry_vivo - m.get("sl", 0))
    r_actual = (upl / (r_dist * m.get("ctVal", 1) * m.get("sz", 1))
                if r_dist > 0 else 0.0)
    if m.get("estado") == "abierta" and r_actual >= TRAILING_DESDE_R:
        pz = abs(int(float(pos.get("pos", 0) or 0)))
        cierre = "sell" if m.get("direccion") == "LONG" else "buy"
        try:
            aid = client.orden_trailing(inst, cierre, TRAILING_CB, pz)
        except Exception as ex:
            aid = None
            registrar("trailing fallo (%s), reintento proximo ciclo" % str(ex)[:60])
        if aid:
            for viejo in (m.get("sl_id"), m.get("tp_id")):
                if viejo:
                    try:
                        client.cancelar_algo(inst, viejo)
                    except Exception:
                        pass
            m["estado"] = "trailing"
            m["trail_id"] = aid
            mg[base] = m
            guardar_json(ARCH_MANAGED, mg)
            avisar("🎗 SCALP %s a %+.1fR -> TRAILING (cb %d%%)\n"
                   "SL y TP retirados: el stop persigue el precio"
                   % (base, r_actual, int(TRAILING_CB * 100)))
    return m


# --------------------------- ciclo ---------------------------
def ciclo(client):
    m = gestionar(client)
    if m is not None:
        return
    if time.time() - ULTIMO_CIERRE_TS < PAUSA_TRAS_CIERRE:
        return
    if os.path.exists(ARCH_PAUSA):
        return
    hoy = trades_hoy()
    if len(hoy) >= MAX_TRADES_DIA:
        return
    for base in MONEDAS:
        if time.time() - COOLDOWNS.get(base, 0) < COOLDOWN_SEG:
            continue
        if perdidas_hoy(base) >= MAX_PERDIDAS_MONEDA:
            continue
        inst = client.inst_id(base)
        if not inst:
            continue
        try:
            d = permiso_1h(velas_cerradas(client, inst, "1H", 250))
            if not d:
                continue
            s = setup_5m(velas_cerradas(client, inst, "5M", 60), d)
            if not s:
                continue
            if gatillo_1m(velas_cerradas(client, inst, "1M", 55), d):
                if abrir(client, base, inst, d, s):
                    return
        except Exception as ex:
            registrar("error analizando %s: %s" % (base, str(ex)[:90]))


# --------------------------- estado ---------------------------
def estado(client):
    print("=== SCALP BOT: estado ===")
    print("Kill switch:", "🛑 PAUSADO (no abre)" if os.path.exists(ARCH_PAUSA)
          else "🟢 activo")
    print("Telegram:", "🟢 configurado" if TG_TOKEN and TG_CHAT else "🔴 sin token")
    try:
        saldo = client.saldo()
        print("Cuenta demo equity: %.2f USD" % float(saldo.get("equity", 0) or 0))
    except Exception as ex:
        print("saldo: no disponible (%s)" % str(ex)[:50])
    m = cargar_json(ARCH_MANAGED, {})
    if m:
        base, v = list(m.items())[0]
        print("Posicion: %s %s %d ct | entry %.6g | SL %.6g | TP %.6g | %s"
              % (base, v.get("direccion"), v.get("sz", 0), v.get("entry", 0),
                 v.get("sl", 0), v.get("tp", 0), v.get("estado", "?")))
        pos = next((p for p in (client.posiciones() or [])
                    if p.get("instId") == v.get("inst")), None)
        if pos:
            print("  P&L flotante: %+.2f USD" % float(pos.get("upl", 0) or 0))
    else:
        print("Posicion: ninguna")
    print("Trades hoy: %d (limite %d)" % (len(trades_hoy()), MAX_TRADES_DIA))
    for j in trades_hoy():
        pnl = j.get("pnl")
        print("  %s %s %s | %s%s" % (
            time.strftime("%H:%M", time.gmtime(j.get("abierta_ts", 0))),
            j.get("moneda"), j.get("direccion"), j.get("estado", "?"),
            (" %+.2f USD" % pnl) if pnl is not None else ""))


# --------------------------- main ---------------------------
def main():
    e = {}
    for l in open("/root/.okx_demo_env"):
        l = l.strip()
        if "=" in l and not l.startswith("#"):
            k, v = l.split("=", 1)
            e[k.replace("export", "").strip()] = v.strip().strip('"').strip("'")
    global TG_TOKEN, TG_CHAT
    TG_TOKEN = e.get("TELEGRAM_BOT_TOKEN", "")
    TG_CHAT = e.get("TELEGRAM_CHAT_ID", "")
    import okx_client as OKX
    client = OKX.Cliente(e.get("OKX_DEMO_KEY", ""), e.get("OKX_DEMO_SECRET", ""),
                         e.get("OKX_DEMO_PASSPHRASE", ""), demo=True)
    if "--estado" in sys.argv:
        estado(client)
        return
    avisar("🤖 Scalp Bot ARRANCA\nMonedas: %s\nRiesgo $%s/trade | max %d/dia | "
           "1 posicion a la vez\nTelegram: %s"
           % (", ".join(MONEDAS), RIESGO_USD, MAX_TRADES_DIA,
              "conectado" if TG_TOKEN and TG_CHAT else "SIN token"))
    n = 0
    while True:
        n += 1
        try:
            ciclo(client)
        except Exception as ex:
            registrar("ERROR en ciclo: " + str(ex)[:120])
        if n % HEARTBEAT_CADA == 0:
            registrar("latido %d | abierta: %s | trades hoy: %d"
                      % (n, "si" if cargar_json(ARCH_MANAGED, {}) else "no",
                         len(trades_hoy())))
        time.sleep(CICLO_SEG)


if __name__ == "__main__":
    main()
