#!/bin/bash
# ENVIAR REPO - sube el diagnostico al repo publico (URL fija, leible por el
# asistente sin limite de peticiones). Reemplaza a enviar_gist.sh.
# Despues de correrlo NO hace falta pegar nada:
#   dile al asistente "mira el diagnostico" y el lo lee directamente.
API="${GIST_API:-https://api.github.com}"
RF="${DIAG_REPO_FILE:-/root/.diag_repo}"
TOF="${GIST_TOKEN_FILE:-/root/.gist_token}"

REPO=""
[ -f "$RF" ] && REPO=$(cat "$RF")
if [ -z "$REPO" ]; then
    echo "FALTA el setup (una vez): bash /root/repo_setup.sh"
    exit 1
fi
TOKEN=""
[ -f "$TOF" ] && TOKEN=$(head -1 "$TOF" | tr -d '[:space:]')
if [ ${#TOKEN} -lt 20 ]; then
    echo "FALTA el token (una vez): bash /root/repo_setup.sh"
    exit 1
fi

case "${GIST_HOST:-$(hostname)}" in
  bot-vm4*)
    NOMBRE="diag_vm4.txt"; RUTA="/root/diagnostico_completo.txt"
    bash /root/recolector.sh > /dev/null 2>&1 ;;
  bot-poly*)
    NOMBRE="diag_poly.txt"; RUTA="/root/diagnostico_poly.txt"
    bash /root/recolector_poly.sh > /dev/null 2>&1 ;;
  *) echo "servidor desconocido: $(hostname)"; exit 1 ;;
esac
[ -f "$RUTA" ] || { echo "el recolector no genero $RUTA"; exit 1; }

python3 - "$TOKEN" "$REPO" "$NOMBRE" "$RUTA" "$API" <<'PYEOF'
import base64, json, sys, urllib.request, urllib.error
token, repo, nombre, ruta, api = sys.argv[1:6]
contenido = open(ruta, encoding="utf-8", errors="replace").read()
if len(contenido) > 900000:
    contenido = contenido[:900000] + "\n... (truncado)"
b64 = base64.b64encode(contenido.encode()).decode()
url = "%s/repos/%s/contents/%s" % (api, repo, nombre)
headers = {"Authorization": "token " + token,
           "Accept": "application/vnd.github+json",
           "User-Agent": "diag-bots"}

sha = None
try:
    req = urllib.request.Request(url, headers=headers)
    existing = json.loads(urllib.request.urlopen(req, timeout=40).read())
    sha = existing.get("sha")
except urllib.error.HTTPError as e:
    if e.code != 404:
        raise SystemExit("fallo comprobando el archivo: " + str(e)[:150])

body = {"message": "diagnostico automatico", "content": b64}
if sha:
    body["sha"] = sha
req = urllib.request.Request(url, data=json.dumps(body).encode(), method="PUT", headers=headers)
try:
    r = urllib.request.urlopen(req, timeout=40)
except Exception as e:
    raise SystemExit("fallo la subida: " + str(e)[:200])
print("HTTP", r.status, "-", nombre, "actualizado")
PYEOF
[ $? -ne 0 ] && exit 1
echo ""
echo "=================================================="
echo "  SUBIDO. Dile al asistente: 'mira el diagnostico'"
echo "  (no hace falta pegar ninguna URL)"
echo "=================================================="
