     6	Estrategia objetiva y medible basada en:
     7	
     8	    AVG7  = media de tweets/día de los últimos 7 días COMPLETOS
     9	    V2    = total de tweets de los últimos 2 días completos
    10	    R     = V2 / (2 × AVG7)                     (ratio de actividad reciente)
    11	    λ48   = 2 × AVG7 × ajuste(R)                (tweets esperados en 48 h)
    12	            ajuste(R) = clamp(1 + 0.5×(R−1), 0.5, 1.5)   (momentum con regresión
    13	                                                          parcial hacia la media)
    14	    X48 ~ Poisson(λ48)                          (modelo de conteo)
    15	
    16	Reglas de entrada (TODAS obligatorias):
    17	    R1  Mercado de ventana 48 h con reglas de resolución claras
    18	        (conteo estándar de estos mercados: posts + quote posts + reposts)
    19	    R2  Volumen ≥ $5.000 y liquidez ≥ $1.000 en el mercado

    24	    R6  UNA sola apuesta activa (secuencial). La siguiente apuesta solo
    25	        se abre cuando la anterior se ha RESUELTO.
    26	    R7  Progresión 3.30 × 1.5^(paso−1), reinicio a 3.30 tras ganar,
    27	        stop-loss de ciclo a los 7 pasos.
    28	
    29	CSV de entrada: columnas  fecha (YYYY-MM-DD), tweets (entero, día completo,

    49	FACTOR          = 1.50          # multiplicador tras cada fallo
    50	CUOTA_MINIMA    = 3.00          # cuota mínima aceptada (precio ≤ 0.333)
    51	P_MIN_YES       = 0.60          # p_modelo mínima para apostar YES
    52	P_MAX_NO        = 0.30          # p_modelo máxima para apostar NO
    53	PASOS_MAX       = 7             # stop-loss del ciclo (paso máximo)

    56	LIQ_MIN         = 1_000         # $ liquidez mínima del mercado
    57	BETA            = 0.5           # coeficiente de momentum del ajuste
    58	AJUSTE_MIN, AJUSTE_MAX = 0.5, 1.5

    61	
    62	def clamp(x, lo, hi):
    63	    return max(lo, min(hi, x))
    64	
    65	
    66	def poisson_cdf(k, lam):
    67	    """P(X ≤ k) con X ~ Poisson(lam). Estable para lam hasta ~700;
    68	    aproximación normal (corrección de continuidad) para lam mayor."""
    69	    if lam <= 0:
    70	        return 1.0 if k >= 0 else 0.0
    71	    if lam > 700:
    72	        z = (k + 0.5 - lam) / math.sqrt(lam)
    73	        return 0.5 * (1.0 + math.erf(z / math.sqrt(2.0)))
    74	    s, term = 0.0, math.exp(-lam)
    75	    kmax = max(0, int(k))
    76	    for i in range(0, kmax + 1):
    77	        s += term
    78	        term *= lam / (i + 1)
    79	    return min(1.0, max(0.0, s))
    80

    82	def p_bin(lo, hi, lam):
    83	    """P(lo ≤ X ≤ hi) con X ~ Poisson(lam). hi puede ser inf."""
    84	    if hi == math.inf:
    85	        return 1.0 - poisson_cdf(lo - 1, lam)
    86	    return poisson_cdf(hi, lam) - poisson_cdf(lo - 1, lam)

    89	def tabla_apuestas(stake_inicial=STAKE_INICIAL, factor=FACTOR, pasos=PASOS_MAX):
    90	    """Genera la tabla de apuestas (stake, pérdida acumulada, beneficio neto)."""
    91	    filas = []
    92	    perdida_prev = 0.0
    93	    for i in range(1, pasos + 1):
    94	        stake = math.ceil(stake_inicial * factor ** (i - 1) * 100) / 100  # redondeo al alza
    95	        perdida_acum = perdida_prev + stake
    96	        # beneficio neto si acierta a cuota 3.00 (pago = stake × 2):
    97	        benef = round(stake * (CUOTA_MINIMA - 1) - perdida_prev, 2)
    98	        filas.append((i, stake, perdida_acum, benef))
    99	        perdida_prev = perdida_acum
   100	    return filas

   117	def metricas(datos):
   118	    """AVG7, V2, R, ajuste, λ48 a partir de los 7/2 últimos días COMPLETOS."""
   119	    ult7 = [t for _, t in datos[-7:]]
   120	    ult2 = [t for _, t in datos[-2:]]
   121	    avg7 = sum(ult7) / 7.0
   122	    v2 = sum(ult2)
   123	    r = v2 / (2.0 * avg7) if avg7 > 0 else float("nan")
   124	    ajuste = clamp(1.0 + BETA * (r - 1.0), AJUSTE_MIN, AJUSTE_MAX) if not math.isnan(r) else 1.0
   125	    lam48 = 2.0 * avg7 * ajuste
   126	    return dict(avg7=avg7, v2=v2, r=r, ajuste=ajuste, lam48=lam48,
   127	                ult7=ult7, ult2=ult2, fecha_base=datos[-1][0])

   130	def decision(m, lo, hi, precio_yes, ya_publicados=0, horas=0):
   131	    """Aplica las reglas R1–R5 y devuelve (p_modelo, cuota_yes, cuota_no,
   132	    veredicto, razon)."""
   133	    if math.isnan(m["r"]):
   134	        return None, None, None, "PASAR", "AVG7 = 0: sin actividad base"
   135	    lam_rest = m["lam48"] * max(0.0, (48.0 - horas)) / 48.0
   136	    p = p_bin(lo - ya_publicados, (hi - ya_publicados) if hi != math.inf else math.inf, lam_rest)
   137	    cuota_yes = 1.0 / precio_yes if precio_yes > 0 else float("inf")
   138	    precio_no = 1.0 - precio_yes
   139	    cuota_no = 1.0 / precio_no if precio_no > 0 else float("inf")
   140	
   141	    razones = []
   142	    if m["avg7"] < AVG7_MIN:
   143	        return p, cuota_yes, cuota_no, "PASAR", f"AVG7 = {m['avg7']:.1f} < {AVG7_MIN} (base insuficiente)"
   144	    if precio_yes > 1 / CUOTA_MINIMA and (1 - precio_yes) > 1 / CUOTA_MINIMA:
   145	        return p, cuota_yes, cuota_no, "PASAR", "Ningún lado cumple cuota ≥ 3.00 (precio > 0.33 en ambos lados)"
   146	    if p >= P_MIN_YES and precio_yes <= 1 / CUOTA_MINIMA:
   147	        return p, cuota_yes, cuota_no, "APOSTAR YES", f"p_modelo {p:.1%} ≥ {P_MIN_YES:.0%} y cuota {cuota_yes:.2f} ≥ 3.00"
   148	    if p <= P_MAX_NO and precio_no <= 1 / CUOTA_MINIMA:
   149	        return p, cuota_yes, cuota_no, "APOSTAR NO", f"p_modelo {p:.1%} ≤ {P_MAX_NO:.0%} y cuota NO {cuota_no:.2f} ≥ 3.00"
   150	    if p >= P_MIN_YES:
   151	        return p, cuota_yes, cuota_no, "PASAR", f"p_modelo {p:.1%} alta pero precio YES {precio_yes:.3f} > 0.33 (cuota < 3)"
   152	    if p <= P_MAX_NO:
   153	        return p, cuota_yes, cuota_no, "PASAR", f"p_modelo {p:.1%} baja pero precio NO {precio_no:.3f} > 0.33 (cuota < 3)"
   154	    return p, cuota_yes, cuota_no, "PASAR", f"p_modelo {p:.1%} sin ventaja (0.30 < p < 0.60)"
   155

   159	    print("=" * 72)

   182	def backtest(datos, lo, hi, precio_yes):
   183	    """Simula la estrategia completa (reglas R1–R7) sobre el histórico.
   184	    Supuestos documentados: entrada al inicio del día t (mercado de 48 h que
   185	    cubre t+1 y t+2), precio de entrada FIJO = precio_yes (sin slippage),
   186	    sin comisiones. El resultado real se calcula con los tweets observados."""
   187	    apuestas = tabla_apuestas()
   188	    ciclo = 0            # paso actual del ciclo (0 = sin ciclo abierto)
   189	    pendiente_hasta = -1 # índice hasta el que hay una apuesta sin resolver
   190	    saldo = 0.0
   191	    max_exp = 0.0
   192	    wins = losses = 0
   193	    racha_p = racha_p_max = 0
   194	    saldo_min = 0.0
   195	    registro = []
   196	
   197	    for t in range(9, len(datos) - 2):
   198	        if t < pendiente_hasta:
   199	            continue
   200	        hist = datos[:t]                      # solo días ya completos
   201	        m = metricas(hist)
   202	        p, c_yes, c_no, veredicto, razon = decision(m, lo, hi, precio_yes)
   203	        if not veredicto.startswith("APOSTAR"):
   204	            continue
   205	        if ciclo == 0:
   206	            ciclo = 1
   207	        paso = ciclo
   208	        _, stake, perd_acum, benef = apuestas[paso - 1]
   209	        max_exp = max(max_exp, perd_acum)
   210	        real = datos[t + 1][1] + datos[t + 2][1]
   211	        ok = (lo <= real <= hi) if hi != math.inf else real >= lo
   212	        if ok:
   213	            saldo += stake * (1 / precio_yes - 1)
   214	            wins += 1
   215	            racha_p = 0
   216	            registro.append((datos[t][0], paso, stake, real, "G", round(saldo, 2)))
   217	            ciclo = 0                          # reinicio del ciclo
   218	        else:
   219	            saldo -= stake
   220	            losses += 1
   221	            racha_p += 1
   222	            racha_p_max = max(racha_p_max, racha_p)
   223	            registro.append((datos[t][0], paso, stake, real, "P", round(saldo, 2)))
   224	            if paso >= PASOS_MAX:
   225	                ciclo = 0                      # stop-loss del ciclo
   226	            else:
   227	                ciclo = paso + 1
   228	        saldo_min = min(saldo_min, saldo)
   229	        pendiente_hasta = t + 3                # resuelve al final de t+2
   230	
   231	    total = wins + losses
   232	    print("=" * 72)
   233	    print("BACKTEST (datos históricos)  ·  bin [%s, %s]  ·  precio YES fijo %.2f" %
   234	          (lo, hi if hi != math.inf else "∞", precio_yes))
   235	    print("=" * 72)
   236	    print(f"  Apuestas totales      : {total}   (ganadas {wins}, perdidas {losses})")
   237	    print(f"  Win rate              : {wins / total:.1%}" if total else "  Win rate: n/d")
   238	    print(f"  Beneficio neto        : ${saldo:+.2f}")
   239	    print(f"  Racha máx de pérdidas : {racha_p_max}")
   240	    print(f"  Exposición máx. ciclo : ${max_exp:.2f}  (máx. teórica paso {PASOS_MAX}: ${apuestas[-1][2]:.2f})")
   241	    print(f"  Peor saldo            : ${saldo_min:.2f}")
   242	    print("\n  Paso a paso (fecha | paso | stake | tweets reales | resultado | saldo):")

   244	        print(f"    {fecha}  paso {paso}  ${stake:>7.2f}  real={real:>4}  {res}  saldo ${sal:>9.2f}")

   277	# === SENAL_EMP_ARENA (27/08): probabilidad empirica (dispersion real) ===
   278	# Sustituye la campana Poisson por la DISTRIBUCION REAL de los ultimos 14
   279	# dias del CSV (convolucion exacta), manteniendo la media con el ajuste de
   280	# momentum. La interfaz no cambia: metricas() y p_bin() reciben y devuelven
   281	# lo mismo, asi que senal_vivo y operar_real funcionan sin tocarlos.
   282	#
   283	# Por que: el Poisson asume una dispersion de +-raiz(lam) (ej. +-9 tweets
   284	# en 48h) cuando Musk se mueve realmente +-40. El modelo compraba
   285	# "certezas" de 60-70% que valian 20-30%.
   286	#
   287	# Interruptor: variable de entorno SENAL_EMP=0 para volver al Poisson
   288	# original al instante (o senal._EMP["on"] = False en codigo/pruebas).
   289	from collections import Counter as _Counter_emp
   290	
   291	try:
   292	    import os as _os_emp
   293	    _EMP_ON_DEF = str(_os_emp.environ.get("SENAL_EMP", "1")).strip().lower() not in ("0", "false", "no", "off")
   294	except Exception:
   295	    _EMP_ON_DEF = True
   296	
   297	_EMP = {"on": _EMP_ON_DEF, "dias": [], "avg7": None, "ajuste": None}
   298	_EMP_N_DIAS = 14
   299	_EMP_CACHE = {}
   300	
   301	
   302	def _emp_dist(k, frac, dias):
   303	    """Distribucion EXACTA de la suma de k dias + frac*otro dia,
   304	    muestreando con reposicion los dias reales (peso 1/n cada dia).
   305	    Convolucion exacta (sin Montecarlo) con cache."""
   306	    clave = (k, round(frac, 4), tuple(dias))
   307	    d = _EMP_CACHE.get(clave)
   308	    if d is not None:
   309	        return d
   310	    nd = float(len(dias))
   311	    base = {}
   312	    for v, c in _Counter_emp(dias).items():
   313	        base[float(v)] = c / nd
   314	    dist = {0.0: 1.0}
   315	    for _ in range(k):
   316	        ndist = {}
   317	        for v1, w1 in dist.items():
   318	            for v2, w2 in base.items():
   319	                v = v1 + v2
   320	                ndist[v] = ndist.get(v, 0.0) + w1 * w2
   321	        dist = ndist
   322	    if frac > 1e-9:
   323	        ndist = {}
   324	        for v1, w1 in dist.items():
   325	            for v2, w2 in base.items():
   326	                v = v1 + v2 * frac
   327	                ndist[v] = ndist.get(v, 0.0) + w1 * w2
   328	        dist = ndist
   329	    if len(_EMP_CACHE) > 64:
   330	        _EMP_CACHE.clear()
   331	    _EMP_CACHE[clave] = dist
   332	    return dist

   354	def p_bin(lo, hi, lam):
   355	    """P(lo <= X <= hi). Empirica si hay estado; Poisson original si no."""
   356	    if not _EMP["on"]:
   357	        return _p_bin_orig(lo, hi, lam)
   358	    try:
   359	        dias = _EMP["dias"]
   360	        avg7 = _EMP["avg7"]
   361	        aj = _EMP["ajuste"]
   362	        if (not dias) or (not avg7) or avg7 <= 0 or (not aj) or aj <= 0 \
   363	                or lam is None or lam <= 0 or hi is None or lo is None:
   364	            return _p_bin_orig(lo, hi, lam)
   365	        base_diaria = avg7 * aj
   366	        dias_eq = lam / base_diaria          # = horas restantes / 24
   367	        k = int(dias_eq)
   368	        frac = dias_eq - k
   369	        dist = _emp_dist(k, frac, dias)
   370	        lo2 = lo / aj
   371	        hi2 = hi / aj                         # hi=inf -> inf/aj = inf (ok)
   372	        p = 0.0
   373	        for v, w in dist.items():
   374	            if lo2 <= v <= hi2:
   375	                p += w
   376	        return p
   377	    except Exception:
   378	        return _p_bin_orig(lo, hi, lam)

   388	# Si el CV (desviacion_tipica/media) de los ultimos 7 dias supera 1.2,
   389	# el modelo empirico es menos fiable. Se acerca p hacia 0.5 para ser
   390	# mas conservador cuando la actividad es muy irregular.
   391	import statistics as _stats_var
   392	CV_MAX_VAR  = 1.2    # umbral de coeficiente de variacion
   393	DESC_VAR    = 0.30   # descuento por unidad de CV excedente

   396	def metricas(datos):
   397	    m = _metricas_var_orig(datos)
   398	    try:
   399	        ult7 = m.get("ult7", [])
   400	        avg7 = m.get("avg7", 0)
   401	        m["cv7"] = (_stats_var.stdev(ult7) / avg7) if (len(ult7) >= 2 and avg7 > 0) else 0.0
   402	    except Exception:
   403	        m["cv7"] = 0.0
   404	    return m

   407	def decision(m, lo, hi, precio_yes, ya_publicados=0, horas=0):
   408	    p, cy, cn, veredicto, razon = _decision_var_orig(m, lo, hi, precio_yes, ya_publicados, horas)
   409	    try:
   410	        cv = m.get("cv7", 0.0)
   411	        if cv > CV_MAX_VAR and p is not None:
   412	            factor = max(0.5, 1.0 - (cv - CV_MAX_VAR) * DESC_VAR)
   413	            p = 0.5 + (p - 0.5) * factor
   414	            if veredicto.startswith("APOSTAR"):
   415	                ok_yes = p >= P_MIN_YES and precio_yes <= 1.0 / CUOTA_MINIMA
   416	                ok_no  = p <= P_MAX_NO  and (1 - precio_yes) <= 1.0 / CUOTA_MINIMA
   417	                if not ok_yes and not ok_no:
   418	                    veredicto = "PASAR"
   419	                    razon = f"FIXVAR: CV={cv:.2f}>{CV_MAX_VAR}, p ajustada {p:.1%} insuficiente"
   420	    except Exception:
   421	        pass
   422	    return p, cy, cn, veredicto, razon

   155	        _MAX=ENTRADA_MAX_H.get(mk["tipo"],12.0)
   156	        _en_inicio=horas_elapsed<=_MAX
   157	        _en_cierre=horas_rest<=_CIERRES.get(mk["tipo"],6.0)
   158	        _en_reentrada=(not _en_inicio and not _en_cierre and horas_rest>_MAX*0.30)
   159	        if not(_en_inicio or _en_cierre or _en_reentrada):
   160	            continue
   161	        _p_min_extra=(0.70 if _en_cierre else 0.68 if _en_reentrada else None)
   162	        if mk["tipo"] == "48h":
   163	            lam_rest = lam48 * max(0.0, horas_rest) / 48.0
   164	        elif mk["tipo"] == "semanal":
   165	            lam_rest = 7 * avg7 * ajuste * max(0.0, horas_rest) / 168.0
   166	        else:
   167	            lam_rest = lam48 * max(0.0, horas_rest) / 48.0
   168	        t0_auto, total_estado = conteo_ventana(inicio, ahora)
   169	        t0 = t0_override if t0_override >= 0 else t0_auto
   170	        bins = []
   171	        for b in mk["bins"]:
   172	            hi = b["hi"] if b["hi"] != float("inf") else math.inf
   173	            # Guarda: bin ya decidido por los tweets publicados (t0).
   174	            #   hi finito y t0 > hi  -> [lo,hi] superado: YES imposible, NO ya decidido
   175	            #   hi = inf  y t0 >= lo -> "≥ lo" ya cumplido: YES ya decidido
   176	            decidido = (t0 > hi) if hi != math.inf else (t0 >= b["lo"])
   177	            if decidido:
   178	                p = 1.0 if hi == math.inf else 0.0
   179	                bins.append({"titulo": b["titulo"], "lo": b["lo"], "hi": b["hi"],
   180	                             "precio_yes": b["precio_yes"], "cuota_yes": b["cuota_yes"],
   181	                             "cuota_no": b["cuota_no"], "p_modelo": p,
   182	                             "veredicto": "PASAR"})
   183	                continue
   184	            p = senal.p_bin(b["lo"] - t0, (hi - t0) if hi != math.inf else math.inf, lam_rest)
   185	            cy, cn = b["cuota_yes"], b["cuota_no"]
   186	            veredicto, lado = "PASAR", None
   187	            if p >= senal.P_MIN_YES and cy and cy >= senal.CUOTA_MINIMA:
   188	                veredicto, lado = "APOSTAR YES", "YES"
   189	            elif p <= senal.P_MAX_NO and cn and cn >= senal.CUOTA_MINIMA:
   190	                veredicto, lado = "APOSTAR NO", "NO"
   191	            bins.append({"titulo": b["titulo"], "lo": b["lo"], "hi": b["hi"],
   192	                         "precio_yes": b["precio_yes"], "cuota_yes": cy,
   193	                         "cuota_no": cn, "p_modelo": p, "veredicto": veredicto})
   194	            if lado:
   195	                cuota_lado = (cy if lado == "YES" else cn)
   196	                p_lado = p if lado == "YES" else (1.0 - p)
   197	                ev = round(p_lado * (cuota_lado or 0), 3)
   198	                stake = stake_motor_v2(stake_base, p_lado, cuota_lado)
   199	                if stake is None:
   200	                    continue  # EV por debajo del mínimo: no entrar
   201	                candidatas.append({"mercado": mk["titulo"], "slug": mk["slug"],

   100	def _descargar_jina(url):
   101	    """Render del perfil vía r.jina.ai. Devuelve {status_id: item}.
   102	
   103	    Acepta tanto el render de x.com como el de twitter.com (perfil en x.com,
   104	    enlaces de estado en twitter.com o x.com). La hora relativa que muestra el
   105	    perfil se convierte a UTC (exacta)."""
   106	    # Cloudflare es intermitente: reintentar con distintos UA y pausas
   107	    uas = [None, UA, "curl/8.0"]
   108	    ultimo_error = None
   109	    for intento in range(4):
   110	        try:
   111	            hdrs = {"Accept": "text/plain"}
   112	            if uas[intento % len(uas)]:
   113	                hdrs["User-Agent"] = uas[intento % len(uas)]
   114	            md = curl(url, hdrs)
   115	            if ("Just a moment" in md or "Verifying your browser" in md
   116	                    or "Rate limit" in md or "[@elonmusk]" not in md):
   117	                ultimo_error = "desafío o respuesta sin contenido"
   118	                time.sleep(8 + intento * 5)
   119	                continue
   120	            ahora = ahora_utc()
   121	            vistos = {}
   122	            # Formato timeline (x.com y twitter.com):
   123	            #   [@elonmusk](https://x.com/elonmusk)  [17h](https://twitter.com/elonmusk/status/ID)
   124	            pat = re.compile(
   125	                r"\[@elonmusk\]\(https://x\.com/elonmusk\)\s+\[([^\]]+)\]"
   126	                r"\(https://(?:twitter\.com|x\.com)/elonmusk/status/(\d+)\)")
   127	            posiciones = list(pat.finditer(md))
   128	            for i, m in enumerate(posiciones):
   129	                rel, sid = m.group(1).strip(), m.group(2)
   130	                ts = rel_a_utc(rel, ahora)
   131	                if not ts:
   132	                    continue
   133	                # trozo de la entrada hasta la siguiente entrada
   134	                fin = posiciones[i + 1].start() if i + 1 < len(posiciones) else len(md)
   135	                trozo = md[m.end():fin]
   136	                # es repost si dentro hay un enlace de estado de OTRO usuario
   137	                es_repost = bool(re.search(
   138	                    r"\]\(https://(?:twitter\.com|x\.com)/(?!elonmusk/status/)"
   139	                    r"[A-Za-z0-9_]+/status/\d+\)", trozo))
   140	                vistos[sid] = {"id": sid,
   141	                               "kind": "repost" if es_repost else "post",
   142	                               "autor": SCREEN,
   143	                               "created_at": ts.strftime(T_FMT),
   144	                               "primera_vista": ahora.strftime(T_FMT),
   145	                               "exacto": True}
   146	            return vistos
   147	        except RuntimeError as e:
   148	            ultimo_error = str(e)
   149	            time.sleep(8 + intento * 5)
   150	    raise RuntimeError(f"jina no disponible tras reintentos: {ultimo_error}")

   242	def rel_a_utc(rel, ahora):
   243	    """Convierte '2h', '3m', '1d', 'Aug 6', 'Aug 6, 2025' a datetime UTC."""
   244	    rel = rel.strip()
   245	    m = re.match(r"^(\d+)([smhd])$", rel)
   246	    if m:
   247	        n, u = int(m.group(1)), m.group(2)
   248	        delta = {"s": 0, "m": 60, "h": 3600, "d": 86400}[u] * n
   249	        return ahora - timedelta(seconds=delta)
   250	    m = re.match(r"^([A-Z][a-z]{2}) (\d{1,2})(?:, (\d{4}))?$", rel)
   251	    if m:
   252	        mes, dia = MESES[m.group(1)], int(m.group(2))
   253	        anio = int(m.group(3)) if m.group(3) else ahora.year
   254	        try:
   255	            ts = datetime(anio, mes, dia, tzinfo=timezone.utc)
   256	        except ValueError:
   257	            return None
   258	        # si la fecha es claramente futura, es del año pasado
   259	        if ts > ahora + timedelta(days=1):
   260	            ts = ts.replace(year=anio - 1)
   261	        return ts
   262	    return None

   121	    """Devuelve dict con proyección o None si no hay datos suficientes.
   122	
   123	    Usa días COMPLETOS (hasta ayer) para el ritmo, para no sobrestimar

   125	    if event_slug not in ventanas:
   126	        return None
   127	    inicio_iso, fin_iso = ventanas[event_slug]
   128	    try:
   129	        ini_dt = datetime.fromisoformat(inicio_iso.replace("Z", "+00:00"))
   130	        fin_dt = datetime.fromisoformat(fin_iso.replace("Z", "+00:00"))
   131	    except Exception:
   132	        return None
   133	    if ini_dt.tzinfo is None:
   134	        ini_dt = ini_dt.replace(tzinfo=timezone.utc)
   135	    if fin_dt.tzinfo is None:
   136	        fin_dt = fin_dt.replace(tzinfo=timezone.utc)
   137	    ahora = datetime.now(timezone.utc)
   138	    dias_totales = max((fin_dt - ini_dt).total_seconds() / 86400, 0.05)
   139	    # días completos transcurridos (hasta ayer)
   140	    hoy = datetime.now(ET).date()
   141	    ini_date = ini_dt.astimezone(ET).date()
   142	    dias_completos = (hoy - ini_date).days
   143	    if dias_completos < 1:
   144	        return None

   146	    tweets_comp = 0
   147	    for fecha, tw in tweets_csv.items():
   148	        try:
   149	            d = datetime.strptime(fecha, "%Y-%m-%d").date()
   150	        except Exception:
   151	            continue
   152	        if ini_date <= d < hoy:
   153	            tweets_comp += tw
   154	    ritmo = tweets_comp / dias_completos
   155	    proyeccion = ritmo * dias_totales
   156	    horas_rest = (fin_dt - ahora).total_seconds() / 3600
   157	    return {
   158	        "tweets_comp": tweets_comp,
   159	        "dias_completos": dias_completos,
   160	        "dias_totales": dias_totales,
   161	        "ritmo": ritmo,

   185	def vender(token_id, shares, cur_price, dry):
   186	    """Vende shares a mercado. Devuelve (ok, detalle)."""
   187	    if dry:
   188	        return True, f"(seco) vendería {shares} shares @ ~{cur_price}"
   189	    client = get_client()
   190	    from py_clob_client_v2.clob_types import MarketOrderArgsV2
   191	    try:
   192	        resp = client.create_and_post_market_order(
   193	            MarketOrderArgsV2(token_id=token_id, amount=shares, side="SELL",
   194	                              order_type="FOK"))
   195	        oid = resp.get("orderID") or resp.get("order_id") or str(resp)[:80]
   196	        return True, f"vendidas {shares} shares (order {oid})"
   197	    except Exception as e:
   198	        # fallback: orden límite al precio actual
   199	        try:
   200	            from py_clob_client_v2.clob_types import OrderArgs
   201	            resp = client.create_and_post_order(
   202	                OrderArgs(token_id=token_id, price=round(cur_price, 4),
   203	                          size=shares, side="SELL"))
   204	            oid = resp.get("orderID") or resp.get("order_id") or str(resp)[:80]
   205	            return True, f"orden límite SELL {shares} @ {cur_price} (order {oid})"
   206	        except Exception as e2:
   207	            return False, f"no se pudo vender: {e} / {e2}"

   210	def evaluar(p, tweets_csv, ventanas, dry):
   211	    """Evalúa una posición y devuelve (accion, texto). accion en
   212	    {'vender', 'mantener', 'sin_datos', 'ultimas_horas'}."""
   213	    bin_ = parse_bin(p.get("title"))
   214	    if bin_ is None:
   215	        return "mantener", "sin bin parseable"
   216	    lo, hi = bin_
   217	    event_slug = p.get("eventSlug", "")
   218	    proy = proyectar(event_slug, tweets_csv, ventanas)
   219	    if proy is None:
   220	        return "mantener", "sin datos de ventana"
   221	    proj = proy["proyeccion"]
   222	    horas = proy["horas_rest"]
   223	    if horas <= MIN_HORAS_RESTANTES:
   224	        return "ultimas_horas", f"quedan {horas:.1f}h (dejar resolver)"
   225	
   226	    fuera_arriba = proj > hi + MARGEN_TWEETS
   227	    fuera_abajo = proj < lo - MARGEN_TWEETS
   228	    if not (fuera_arriba or fuera_abajo):
   229	        return "mantener", (f"proyección {proj:.1f} dentro de [{lo},{hi}]")
   230	
   231	    # hay que cerrar
   232	    side = p.get("outcome", "Yes")
   233	    if side == "No":
   234	        # para NO, gana si NO cae en el bin: fuera es BUENO, dentro es MALO
   235	        return "mantener", "posición NO (no se gestiona)"
   236	    cur = float(p.get("curPrice", 0) or 0)
   237	    avg = float(p.get("avgPrice", 0) or 0)
   238	    pnl_pct = (cur / avg - 1) * 100 if avg > 0 else 0
   239	    shares = float(p.get("size", 0) or 0)
   240	    token_id = p.get("asset", "")
   241	    razon = (f"se pasa por arriba ({proj:.0f}>{hi})" if fuera_arriba
   242	             else f"no llega al mínimo ({proj:.0f}<{lo})")
   243	    if not token_id or shares <= 0:
   244	        return "mantener", f"sin token/tamaño (token={bool(token_id)})"
   245	    ok, detalle = vender(token_id, shares, cur, dry)
   246	    init_val = float(p.get("initialValue", 0) or 0)
   247	    cur_val = float(p.get("currentValue", 0) or 0)
   248	    pnl = cur_val - init_val
   249	    ic_bal = "🟢" if pnl >= 0 else "🔴"
   250	    txt = (f"{'🧪 SECO' if dry else '🔻 CIERRE ANTICIPADO'} {p.get('title')}\n"
   251	           f"   bin [{lo}-{hi}] · proyección {proj:.0f} → {razon}\n"
   252	           f"   {shares:.1f} shares · entrada {avg:.4f} · ahora {cur:.4f} "
   253	           f"({pnl_pct:+.0f}%)\n"
   254	           f"   💰 Balance de la operación: {ic_bal} ${pnl:+.2f} "
   255	           f"(invertido ${init_val:.2f} → ahora ${cur_val:.2f})\n"
   256	           f"   → {detalle}")
   257	    if ok and not dry:
   258	        _guardar_cierre({
   259	            "fecha": datetime.now(ET).strftime("%Y-%m-%d"),
   260	            "titulo": (p.get("title") or "?").replace("Elon Musk # tweets ", "").replace("?", ""),
   261	            "bin": f"{lo}-{hi}",
   262	            "lado": p.get("outcome", "Yes"),
   263	            "invertido": round(init_val, 2),
   264	            "valor": round(cur_val, 2),
   265	            "pnl": round(pnl, 2),
   266	        })
   267	    if ok:
   268	        tg_send(txt)
   269	    return ("vender" if ok else "fallo"), txt

   330	        except FileNotFoundError:

   417	    import re as _re
   418	    toks = tokens_de_slug(slug)
   419	    nums = []
   420	    for tok in toks:

   426	    except Exception:

   428	    for c in d.get("cierres", []):
   429	        t = (c.get("titulo") or "").lower()
   430	        if bin_t.lower() not in t:
   431	            continue
   432	        tnums = _re.findall(r"\d+", t)
   433	        if nums and all(n in tnums for n in nums):
   434	            return float(c.get("pnl", 0) or 0)
   435	        if any(m in t for m in MESES_ES):

   440	def arreglar_atascado(nombre, path, act):
   441	    slug = act.get("slug", "")
   442	    bin_t = act.get("bin_titulo", "")
   443	    lado = act.get("lado", "YES")
   444	    # === FIXZELB_ARENA: apuesta PENDIENTE -> mirar el llenado REAL antes de nada ===
   445	    if act.get("pendiente") and act.get("order_id"):
   446	        _szb = 0.0
   447	        try:
   448	            sys.path.insert(0, f"{BASE}/bot-polymarket-elon")
   449	            import operar_real as _opb
   450	            _detb = _opb.get_client().get_order(act["order_id"])
   451	            if _detb:
   452	                _szb = float(_detb.get("size_matched", 0) or 0)
   453	        except Exception:
   454	            _szb = 0.0
   455	        if _szb > 0 and act.get("precio"):
   456	            act["size"] = round(_szb, 4)
   457	            act["stake"] = round(_szb * float(act.get("precio") or 0), 2)
   458	            act["pendiente"] = False
   459	            try:
   460	                _estb = json.load(open(path, encoding="utf-8"))
   461	                if _estb.get("activa"):
   462	                    _estb["activa"]["size"] = act["size"]
   463	                    _estb["activa"]["stake"] = act["stake"]
   464	                    _estb["activa"]["pendiente"] = False
   465	                    json.dump(_estb, open(path, "w", encoding="utf-8"),
   466	                              ensure_ascii=False, indent=1)
   467	            except Exception:
   468	                pass
   469	    stake = float(act.get("stake", 0) or 0)
   470	    cuota = float(act.get("cuota", 0) or 0)
   471	
   472	    cerrado, ganador = gamma_info(slug)
   473	
   474	    # Caso especial: orden "pendiente" que NUNCA se llenó (el mercado sigue
   475	    # abierto y la orden ya no está en CLOB, cancelada o expirada). NO hay
   476	    # pérdida real: solo limpiamos el estado y liberamos la ventana.
   477	    if not cerrado and act.get("pendiente"):
   478	        no_llena = False
   479	        if act.get("order_id"):
   480	            try:
   481	                sys.path.insert(0, f"{BASE}/bot-polymarket-elon")
   482	                import operar_real as _op
   483	                det = _op.get_client().get_order(act["order_id"])
   484	                st = (det.get("status") or "").lower() if det else ""
   485	                sz = float(det.get("size_matched", 0) or 0) if det else 0.0
   486	                llenada = st in ("matched", "filled") or sz > 0
   487	                no_llena = (det is None) or (not llenada and st in
   488	                            ("cancelled", "canceled", "expired",
   489	                             "cancelled_by_user"))
   490	            except Exception:
   491	                no_llena = False
   492	        if no_llena:
   493	            try:
   494	                estado = json.load(open(path, encoding="utf-8"))
   495	            except Exception:
   496	                return f"{nombre}: no pude leer {path}"
   497	            estado["activa"] = None
   498	            json.dump(estado, open(path, "w", encoding="utf-8"),
   499	                      ensure_ascii=False, indent=1)
   500	            try:
   501	                lock = json.load(open(LOCK, encoding="utf-8"))
   502	            except Exception:
   503	                lock = {}
   504	            lock.pop(f"{slug}|{bin_t}", None)
   505	            json.dump(lock, open(LOCK, "w", encoding="utf-8"),
   506	                      ensure_ascii=False, indent=1)
   507	            return (f"{nombre}: orden '{bin_t}' {lado} nunca se llenó "
   508	                    f"(cancelada/expirada) → estado limpiado, sin pérdida")
   509	
   510	    res = None
   511	    detalle = ""
   512	    if cerrado and ganador:
   513	        gana = (ganador.lower() == bin_t.lower()) if lado == "YES" else \
   514	               (ganador.lower() != bin_t.lower())
   515	        res = "G" if gana else "P"
   516	        detalle = f"resuelto por Polymarket: ganó {ganador}"
   517	    else:
   518	        pnl = cierre_anticipado(slug, bin_t)
   519	        if pnl is not None and pnl > 0:
   520	            res = "G"
   521	            detalle = f"vendido por el gestor en positivo ({pnl:+.2f} a nivel cuenta)"
   522	        else:

   525	                       else "vendido/desaparecido (sin registro del gestor)")
   526	
   527	    try:
   528	        estado = json.load(open(path, encoding="utf-8"))
   529	    except Exception:
   530	        return f"{nombre}: no pude leer {path}"
   531	
   532	    if res == "G":
   533	        benef = round(stake * (cuota - 1), 2) if cuota else round(stake, 2)
   534	        paso = 1
   535	    else:
   536	        benef = -round(stake, 2)
   537	        paso_ant = int(act.get("paso", 1))
   538	        paso = 1 if paso_ant >= 7 else paso_ant + 1
   539	    estado["saldo"] = round(float(estado.get("saldo", 500.0)) + benef, 2)

   545	        "paso": int(act.get("paso", 1)), "stake": round(stake, 2),
   546	        "real": detalle, "resultado": res, "beneficio": benef,
   547	        "saldo": round(estado["saldo"], 2),

   549	    estado.setdefault("historial", []).append(reg)

   554	        lock = json.load(open(LOCK, encoding="utf-8"))

   558	    json.dump(lock, open(LOCK, "w", encoding="utf-8"), ensure_ascii=False, indent=1)
   559	    return f"{nombre}: reconciliado '{bin_t}' → {res} ({benef:+.2f}) · paso {paso} · {detalle}"
