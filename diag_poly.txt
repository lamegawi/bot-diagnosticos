    62	def clamp(x, lo, hi):
    63	    return max(lo, min(hi, x))

    66	def poisson_cdf(k, lam):
    67	    """P(X ≤ k) con X ~ Poisson(lam). Estable para lam hasta ~700;
    68	    aproximación normal (corrección de continuidad) para lam mayor."""
    69	    if lam <= 0:
    70	        return 1.0 if k >= 0 else 0.0
    71	    if lam > 700:
    72	        z = (k + 0.5 - lam) / math.sqrt(lam)
    73	        return 0.5 * (1.0 + math.erf(z / math.sqrt(2.0)))
    74	    s, term = 0.0, math.exp(-lam)
    75	    kmax = max(0, int(k))
    76	    for i in range(0, kmax + 1):
    77	        s += term
    78	        term *= lam / (i + 1)
    79	    return min(1.0, max(0.0, s))

    82	def p_bin(lo, hi, lam):
    83	    """P(lo ≤ X ≤ hi) con X ~ Poisson(lam). hi puede ser inf."""
    84	    if hi == math.inf:
    85	        return 1.0 - poisson_cdf(lo - 1, lam)
    86	    return poisson_cdf(hi, lam) - poisson_cdf(lo - 1, lam)

   117	def metricas(datos):
   118	    """AVG7, V2, R, ajuste, λ48 a partir de los 7/2 últimos días COMPLETOS."""
   119	    ult7 = [t for _, t in datos[-7:]]
   120	    ult2 = [t for _, t in datos[-2:]]
   121	    avg7 = sum(ult7) / 7.0
   122	    v2 = sum(ult2)
   123	    r = v2 / (2.0 * avg7) if avg7 > 0 else float("nan")
   124	    ajuste = clamp(1.0 + BETA * (r - 1.0), AJUSTE_MIN, AJUSTE_MAX) if not math.isnan(r) else 1.0
   125	    lam48 = 2.0 * avg7 * ajuste
   126	    return dict(avg7=avg7, v2=v2, r=r, ajuste=ajuste, lam48=lam48,
   127	                ult7=ult7, ult2=ult2, fecha_base=datos[-1][0])

   138	    precio_no = 1.0 - precio_yes

   141	    razones = []

   182	def backtest(datos, lo, hi, precio_yes):
   183	    """Simula la estrategia completa (reglas R1–R7) sobre el histórico.
   184	    Supuestos documentados: entrada al inicio del día t (mercado de 48 h que
   185	    cubre t+1 y t+2), precio de entrada FIJO = precio_yes (sin slippage),
   186	    sin comisiones. El resultado real se calcula con los tweets observados."""
   187	    apuestas = tabla_apuestas()
   188	    ciclo = 0            # paso actual del ciclo (0 = sin ciclo abierto)
   189	    pendiente_hasta = -1 # índice hasta el que hay una apuesta sin resolver
   190	    saldo = 0.0
   191	    max_exp = 0.0
   192	    wins = losses = 0
   193	    racha_p = racha_p_max = 0
   194	    saldo_min = 0.0
   195	    registro = []
   196	
   197	    for t in range(9, len(datos) - 2):
   198	        if t < pendiente_hasta:
   199	            continue
   200	        hist = datos[:t]                      # solo días ya completos
   201	        m = metricas(hist)
   202	        p, c_yes, c_no, veredicto, razon = decision(m, lo, hi, precio_yes)
   203	        if not veredicto.startswith("APOSTAR"):
   204	            continue
   205	        if ciclo == 0:
   206	            ciclo = 1
   207	        paso = ciclo
   208	        _, stake, perd_acum, benef = apuestas[paso - 1]
   209	        max_exp = max(max_exp, perd_acum)
   210	        real = datos[t + 1][1] + datos[t + 2][1]
   211	        ok = (lo <= real <= hi) if hi != math.inf else real >= lo
   212	        if ok:
   213	            saldo += stake * (1 / precio_yes - 1)
   214	            wins += 1
   215	            racha_p = 0
   216	            registro.append((datos[t][0], paso, stake, real, "G", round(saldo, 2)))
   217	            ciclo = 0                          # reinicio del ciclo
   218	        else:
   219	            saldo -= stake
   220	            losses += 1
   221	            racha_p += 1
   222	            racha_p_max = max(racha_p_max, racha_p)
   223	            registro.append((datos[t][0], paso, stake, real, "P", round(saldo, 2)))
   224	            if paso >= PASOS_MAX:
   225	                ciclo = 0                      # stop-loss del ciclo
   226	            else:
   227	                ciclo = paso + 1
   228	        saldo_min = min(saldo_min, saldo)
   229	        pendiente_hasta = t + 3                # resuelve al final de t+2
   230	
   231	    total = wins + losses
   232	    print("=" * 72)
   233	    print("BACKTEST (datos históricos)  ·  bin [%s, %s]  ·  precio YES fijo %.2f" %
   234	          (lo, hi if hi != math.inf else "∞", precio_yes))
   235	    print("=" * 72)
   236	    print(f"  Apuestas totales      : {total}   (ganadas {wins}, perdidas {losses})")
   237	    print(f"  Win rate              : {wins / total:.1%}" if total else "  Win rate: n/d")
   238	    print(f"  Beneficio neto        : ${saldo:+.2f}")
   239	    print(f"  Racha máx de pérdidas : {racha_p_max}")
   240	    print(f"  Exposición máx. ciclo : ${max_exp:.2f}  (máx. teórica paso {PASOS_MAX}: ${apuestas[-1][2]:.2f})")
   241	    print(f"  Peor saldo            : ${saldo_min:.2f}")
   242	    print("\n  Paso a paso (fecha | paso | stake | tweets reales | resultado | saldo):")
   243	    for fecha, paso, stake, real, res, sal in registro:
   244	        print(f"    {fecha}  paso {paso}  ${stake:>7.2f}  real={real:>4}  {res}  saldo ${sal:>9.2f}")
   245	    return registro

   277	# === SENAL_EMP_ARENA (27/08): probabilidad empirica (dispersion real) ===
   278	# Sustituye la campana Poisson por la DISTRIBUCION REAL de los ultimos 14
   279	# dias del CSV (convolucion exacta), manteniendo la media con el ajuste de
   280	# momentum. La interfaz no cambia: metricas() y p_bin() reciben y devuelven
   281	# lo mismo, asi que senal_vivo y operar_real funcionan sin tocarlos.
   282	#
   283	# Por que: el Poisson asume una dispersion de +-raiz(lam) (ej. +-9 tweets
   284	# en 48h) cuando Musk se mueve realmente +-40. El modelo compraba
   285	# "certezas" de 60-70% que valian 20-30%.
   286	#
   287	# Interruptor: variable de entorno SENAL_EMP=0 para volver al Poisson
   288	# original al instante (o senal._EMP["on"] = False en codigo/pruebas).
   289	from collections import Counter as _Counter_emp
   290	
   291	try:
   292	    import os as _os_emp
   293	    _EMP_ON_DEF = str(_os_emp.environ.get("SENAL_EMP", "1")).strip().lower() not in ("0", "false", "no", "off")
   294	except Exception:
   295	    _EMP_ON_DEF = True

   297	_EMP = {"on": _EMP_ON_DEF, "dias": [], "avg7": None, "ajuste": None}
   298	_EMP_N_DIAS = 14
   299	_EMP_CACHE = {}

   301	
   302	def _emp_dist(k, frac, dias):
   303	    """Distribucion EXACTA de la suma de k dias + frac*otro dia,
   304	    muestreando con reposicion los dias reales (peso 1/n cada dia).
   305	    Convolucion exacta (sin Montecarlo) con cache."""
   306	    clave = (k, round(frac, 4), tuple(dias))
   307	    d = _EMP_CACHE.get(clave)
   308	    if d is not None:
   309	        return d
   310	    nd = float(len(dias))
   311	    base = {}
   312	    for v, c in _Counter_emp(dias).items():
   313	        base[float(v)] = c / nd
   314	    dist = {0.0: 1.0}
   315	    for _ in range(k):
   316	        ndist = {}
   317	        for v1, w1 in dist.items():
   318	            for v2, w2 in base.items():
   319	                v = v1 + v2
   320	                ndist[v] = ndist.get(v, 0.0) + w1 * w2
   321	        dist = ndist
   322	    if frac > 1e-9:
   323	        ndist = {}
   324	        for v1, w1 in dist.items():
   325	            for v2, w2 in base.items():
   326	                v = v1 + v2 * frac
   327	                ndist[v] = ndist.get(v, 0.0) + w1 * w2
   328	        dist = ndist
   329	    if len(_EMP_CACHE) > 64:
   330	        _EMP_CACHE.clear()
   331	    _EMP_CACHE[clave] = dist
   332	    return dist

   355	    """P(lo <= X <= hi). Empirica si hay estado; Poisson original si no."""
   356	    if not _EMP["on"]:
   357	        return _p_bin_orig(lo, hi, lam)
   358	    try:
   359	        dias = _EMP["dias"]
   360	        avg7 = _EMP["avg7"]
   361	        aj = _EMP["ajuste"]
   362	        if (not dias) or (not avg7) or avg7 <= 0 or (not aj) or aj <= 0 \
   363	                or lam is None or lam <= 0 or hi is None or lo is None:
   364	            return _p_bin_orig(lo, hi, lam)
   365	        base_diaria = avg7 * aj
   366	        dias_eq = lam / base_diaria          # = horas restantes / 24
   367	        k = int(dias_eq)
   368	        frac = dias_eq - k
   369	        dist = _emp_dist(k, frac, dias)
   370	        lo2 = lo / aj
   371	        hi2 = hi / aj                         # hi=inf -> inf/aj = inf (ok)
   372	        p = 0.0
   373	        for v, w in dist.items():
   374	            if lo2 <= v <= hi2:
   375	                p += w
   376	        return p
   377	    except Exception:
   378	        return _p_bin_orig(lo, hi, lam)

   133	def evaluar(avg7, v2, ajuste, lam48, mercados, paso, t0_override=-1, ahora=None):
   134	    """Evalúa todos los mercados abiertos y devuelve (evaluados, candidatas).

   166	            # Guarda: bin ya decidido por los tweets publicados (t0).

   170	            if decidido:

   175	                             "veredicto": "PASAR"})
   176	                continue
   177	            p = senal.p_bin(b["lo"] - t0, (hi - t0) if hi != math.inf else math.inf, lam_rest)
   178	            cy, cn = b["cuota_yes"], b["cuota_no"]
   179	            veredicto, lado = "PASAR", None
   180	            if p >= senal.P_MIN_YES and cy and cy >= senal.CUOTA_MINIMA:
   181	                veredicto, lado = "APOSTAR YES", "YES"
   182	            elif p <= senal.P_MAX_NO and cn and cn >= senal.CUOTA_MINIMA:
   183	                veredicto, lado = "APOSTAR NO", "NO"
   184	            bins.append({"titulo": b["titulo"], "lo": b["lo"], "hi": b["hi"],
   185	                         "precio_yes": b["precio_yes"], "cuota_yes": cy,
   186	                         "cuota_no": cn, "p_modelo": p, "veredicto": veredicto})
   187	            if lado:
   188	                cuota_lado = (cy if lado == "YES" else cn)
   189	                p_lado = p if lado == "YES" else (1.0 - p)
   190	                ev = round(p_lado * (cuota_lado or 0), 3)
   191	                stake = stake_motor_v2(stake_base, p_lado, cuota_lado)
   192	                if stake is None:
   193	                    continue  # EV por debajo del mínimo: no entrar
   194	                candidatas.append({"mercado": mk["titulo"], "slug": mk["slug"],
   195	                                   "ventana": (inicio, fin), "tipo": mk["tipo"],
   196	                                   "bin_titulo": b["titulo"], "lo": b["lo"], "hi": b["hi"],
   197	                                   "lado": lado, "precio": (b["precio_yes"] if lado == "YES" else 1 - b["precio_yes"]),
   198	                                   "cuota": cuota_lado, "p_modelo": p,

   201	                          "inicio": inicio, "fin": fin, "horas_rest": horas_rest,
   202	                          "lam_rest": lam_rest, "t0": t0, "bins": bins})
   203	    return evaluados, candidatas

    88	def curl(url, headers=None, timeout=60):
    89	    cmd = ["curl", "-s", "--max-time", str(timeout), "-L", url]
    90	    for k, v in (headers or {}).items():
    91	        cmd += ["-H", f"{k}: {v}"]
    92	    r = subprocess.run(cmd, capture_output=True, text=True, encoding="utf-8", errors="replace")
    93	    if r.returncode != 0:
    94	        raise RuntimeError(f"curl falló ({r.returncode}): {r.stderr[:200]}")
    95	    return r.stdout

   100	def _descargar_jina(url):
   101	    """Render del perfil vía r.jina.ai. Devuelve {status_id: item}.
   102	
   103	    Acepta tanto el render de x.com como el de twitter.com (perfil en x.com,
   104	    enlaces de estado en twitter.com o x.com). La hora relativa que muestra el
   105	    perfil se convierte a UTC (exacta)."""
   106	    # Cloudflare es intermitente: reintentar con distintos UA y pausas
   107	    uas = [None, UA, "curl/8.0"]
   108	    ultimo_error = None
   109	    for intento in range(4):
   110	        try:
   111	            hdrs = {"Accept": "text/plain"}
   112	            if uas[intento % len(uas)]:
   113	                hdrs["User-Agent"] = uas[intento % len(uas)]
   114	            md = curl(url, hdrs)
   115	            if ("Just a moment" in md or "Verifying your browser" in md
   116	                    or "Rate limit" in md or "[@elonmusk]" not in md):
   117	                ultimo_error = "desafío o respuesta sin contenido"
   118	                time.sleep(8 + intento * 5)
   119	                continue
   120	            ahora = ahora_utc()
   121	            vistos = {}
   122	            # Formato timeline (x.com y twitter.com):
   123	            #   [@elonmusk](https://x.com/elonmusk)  [17h](https://twitter.com/elonmusk/status/ID)
   124	            pat = re.compile(
   125	                r"\[@elonmusk\]\(https://x\.com/elonmusk\)\s+\[([^\]]+)\]"
   126	                r"\(https://(?:twitter\.com|x\.com)/elonmusk/status/(\d+)\)")
   127	            posiciones = list(pat.finditer(md))
   128	            for i, m in enumerate(posiciones):
   129	                rel, sid = m.group(1).strip(), m.group(2)
   130	                ts = rel_a_utc(rel, ahora)
   131	                if not ts:
   132	                    continue
   133	                # trozo de la entrada hasta la siguiente entrada
   134	                fin = posiciones[i + 1].start() if i + 1 < len(posiciones) else len(md)
   135	                trozo = md[m.end():fin]
   136	                # es repost si dentro hay un enlace de estado de OTRO usuario
   137	                es_repost = bool(re.search(
   138	                    r"\]\(https://(?:twitter\.com|x\.com)/(?!elonmusk/status/)"
   139	                    r"[A-Za-z0-9_]+/status/\d+\)", trozo))
   140	                vistos[sid] = {"id": sid,
   141	                               "kind": "repost" if es_repost else "post",
   142	                               "autor": SCREEN,
   143	                               "created_at": ts.strftime(T_FMT),
   144	                               "primera_vista": ahora.strftime(T_FMT),
   145	                               "exacto": True}
   146	            return vistos
   147	        except RuntimeError as e:
   148	            ultimo_error = str(e)
   149	            time.sleep(8 + intento * 5)
   150	    raise RuntimeError(f"jina no disponible tras reintentos: {ultimo_error}")

   231	def parse_fecha_nitter(titulo):
   232	    m = re.match(r"([A-Z][a-z]{2} \d{1,2}, \d{4}) · (\d{1,2}:\d{2} [AP]M) UTC", titulo)
   233	    if not m:
   234	        return None
   235	    try:
   236	        return datetime.strptime(f"{m.group(1)} {m.group(2)}", "%b %d, %Y %I:%M %p") \
   237	            .replace(tzinfo=timezone.utc)
   238	    except ValueError:
   239	        return None

   242	def rel_a_utc(rel, ahora):
   243	    """Convierte '2h', '3m', '1d', 'Aug 6', 'Aug 6, 2025' a datetime UTC."""
   244	    rel = rel.strip()
   245	    m = re.match(r"^(\d+)([smhd])$", rel)
   246	    if m:
   247	        n, u = int(m.group(1)), m.group(2)
   248	        delta = {"s": 0, "m": 60, "h": 3600, "d": 86400}[u] * n
   249	        return ahora - timedelta(seconds=delta)
   250	    m = re.match(r"^([A-Z][a-z]{2}) (\d{1,2})(?:, (\d{4}))?$", rel)
   251	    if m:
   252	        mes, dia = MESES[m.group(1)], int(m.group(2))
   253	        anio = int(m.group(3)) if m.group(3) else ahora.year
   254	        try:
   255	            ts = datetime(anio, mes, dia, tzinfo=timezone.utc)
   256	        except ValueError:
   257	            return None
   258	        # si la fecha es claramente futura, es del año pasado
   259	        if ts > ahora + timedelta(days=1):
   260	            ts = ts.replace(year=anio - 1)
   261	        return ts
   262	    return None

   267	def conteo_diario(vistos, sin_reposts=False, modo_loop=False):
   268	    """Agrupa por día ET.
   269	    - Items con timestamp exacto (jina/x.com): created_at siempre.
   270	    - Reposts de xcancel (fecha del original): primera_vista en modo_loop
   271	      (≈ momento real, error < intervalo); si no, fecha del original."""
   272	    dias = {}
   273	    for v in vistos.values():
   274	        if v.get("kind") == "repost" and sin_reposts:
   275	            continue
   276	        if v.get("exacto") or v.get("kind") != "repost" or not modo_loop:
   277	            base = v["created_at"]
   278	        else:
   279	            base = v["primera_vista"]
   280	        fecha = datetime.strptime(base, T_FMT).astimezone(ET).date()
   281	        dias[fecha] = dias.get(fecha, 0) + 1
   282	    return dias

   285	def actualizar_csv(dias_conteo, completos=True):
   286	    hoy = hoy_et()
   287	    filas = {}
   288	    if os.path.exists(CSV):
   289	        with open(CSV, newline="", encoding="utf-8") as f:
   290	            for fila in csv.DictReader(f):
   291	                filas[fila["fecha"]] = int(fila["tweets"])
   292	    nuevos = 0
   293	    for fecha, n in sorted(dias_conteo.items()):
   294	        if completos and fecha >= hoy:
   295	            continue
   296	        # MONÓTONO: el conteo directo es una cota inferior (subconteo parcial);
   297	        # solo se actualiza si el nuevo valor es MAYOR. Así los días
   298	        # reconstruidos desde los mercados resueltos no se degradan.
   299	        if n > filas.get(fecha.isoformat(), -1):
   300	            filas[fecha.isoformat()] = n
   301	            nuevos += 1
   302	    with open(CSV, "w", newline="", encoding="utf-8") as f:
   303	        w = csv.writer(f)
   304	        w.writerow(["fecha", "tweets"])
   305	        for fecha in sorted(filas):
   306	            w.writerow([fecha, filas[fecha]])
   307	    return nuevos, len(filas)

   329	def fusionar(vistos_nuevos, sin_reposts=False):
   330	    """Fusiona con el estado previo: por status id, gana el item nuevo
   331	    (jina/x.com da el tiempo exacto del repost) pero se CONSERVA la
   332	    primera_vista más antigua (el repost se atribuye al día en que se
   333	    vio por PRIMERA vez). Si sin_reposts, elimina los reposts."""
   334	    estado = cargar_estado()
   335	    if sin_reposts:
   336	        estado = {sid: v for sid, v in estado.items() if v.get("kind") != "repost"}
   337	    for sid, v in vistos_nuevos.items():
   338	        previo = estado.get(sid)
   339	        if previo:
   340	            v["primera_vista"] = min(v["primera_vista"], previo["primera_vista"])
   341	        estado[sid] = v
   342	    return estado

   349	    with open(CSV, newline="", encoding="utf-8") as f:

   352	    print(f"{'fecha':<12}{'tweets':>7}")
   353	    for fecha, n in filas[-dias:]:
   354	        marca = "  ← hoy (incompleto, no entra en la señal)" if fecha == hoy else ""
   355	        print(f"{fecha:<12}{n:>7}{marca}")
   356	    if len(filas) >= 9:
   357	        ult7 = [n for _, n in filas[-7:]]
   358	        ult2 = [n for _, n in filas[-2:]]
   359	        avg7 = sum(ult7) / 7
   360	        v2 = sum(ult2)
   361	        r = v2 / (2 * avg7)
   362	        aj = min(1.5, max(0.5, 1 + 0.5 * (r - 1)))
   363	        lam = 2 * avg7 * aj
   364	        print(f"\nAVG7 = {avg7:.2f}  V2 = {v2}  R = {r:.3f}  ajuste = {aj:.3f}  λ48 = {lam:.1f}")
   365	    print("Días completos en CSV:", len(filas))

   210	def evaluar(p, tweets_csv, ventanas, dry):
   211	    """Evalúa una posición y devuelve (accion, texto). accion en
   212	    {'vender', 'mantener', 'sin_datos', 'ultimas_horas'}."""
   213	    bin_ = parse_bin(p.get("title"))
   214	    if bin_ is None:
   215	        return "mantener", "sin bin parseable"
   216	    lo, hi = bin_
   217	    event_slug = p.get("eventSlug", "")
   218	    proy = proyectar(event_slug, tweets_csv, ventanas)
   219	    if proy is None:
   220	        return "mantener", "sin datos de ventana"
   221	    proj = proy["proyeccion"]
   222	    horas = proy["horas_rest"]
   223	    if horas <= MIN_HORAS_RESTANTES:
   224	        return "ultimas_horas", f"quedan {horas:.1f}h (dejar resolver)"
   225	
   226	    fuera_arriba = proj > hi + MARGEN_TWEETS
   227	    fuera_abajo = proj < lo - MARGEN_TWEETS
   228	    if not (fuera_arriba or fuera_abajo):
   229	        return "mantener", (f"proyección {proj:.1f} dentro de [{lo},{hi}]")
   230	
   231	    # hay que cerrar
   232	    side = p.get("outcome", "Yes")
   233	    if side == "No":
   234	        # para NO, gana si NO cae en el bin: fuera es BUENO, dentro es MALO
   235	        return "mantener", "posición NO (no se gestiona)"
   236	    cur = float(p.get("curPrice", 0) or 0)
   237	    avg = float(p.get("avgPrice", 0) or 0)
   238	    pnl_pct = (cur / avg - 1) * 100 if avg > 0 else 0
   239	    shares = float(p.get("size", 0) or 0)
   240	    token_id = p.get("asset", "")
   241	    razon = (f"se pasa por arriba ({proj:.0f}>{hi})" if fuera_arriba
   242	             else f"no llega al mínimo ({proj:.0f}<{lo})")
   243	    if not token_id or shares <= 0:
   244	        return "mantener", f"sin token/tamaño (token={bool(token_id)})"
   245	    ok, detalle = vender(token_id, shares, cur, dry)
   246	    init_val = float(p.get("initialValue", 0) or 0)
   247	    cur_val = float(p.get("currentValue", 0) or 0)
   248	    pnl = cur_val - init_val
   249	    ic_bal = "🟢" if pnl >= 0 else "🔴"
   250	    txt = (f"{'🧪 SECO' if dry else '🔻 CIERRE ANTICIPADO'} {p.get('title')}\n"
   251	           f"   bin [{lo}-{hi}] · proyección {proj:.0f} → {razon}\n"
   252	           f"   {shares:.1f} shares · entrada {avg:.4f} · ahora {cur:.4f} "
   253	           f"({pnl_pct:+.0f}%)\n"
   254	           f"   💰 Balance de la operación: {ic_bal} ${pnl:+.2f} "
   255	           f"(invertido ${init_val:.2f} → ahora ${cur_val:.2f})\n"
   256	           f"   → {detalle}")
   257	    if ok and not dry:
   258	        _guardar_cierre({
   259	            "fecha": datetime.now(ET).strftime("%Y-%m-%d"),
   260	            "titulo": (p.get("title") or "?").replace("Elon Musk # tweets ", "").replace("?", ""),
   261	            "bin": f"{lo}-{hi}",
   262	            "lado": p.get("outcome", "Yes"),
   263	            "invertido": round(init_val, 2),
   264	            "valor": round(cur_val, 2),
   265	            "pnl": round(pnl, 2),
   266	        })
   267	    if ok:
   268	        tg_send(txt)
   269	    return ("vender" if ok else "fallo"), txt

   112	    cmd = ["curl", "-s", "--max-time", "35"]
   113	    if proxy:
   114	        cmd += ["-x", PROXY]
   115	    cmd += [url]
   116	    r = subprocess.run(cmd, capture_output=True, text=True,
   117	                       encoding="utf-8", errors="replace")
   118	    if r.returncode != 0 or not r.stdout.strip():
   119	        return None
   120	    try:
   121	        return json.loads(r.stdout)
   122	    except Exception:
   123	        return None

   246	def _orden_viva(order_id):
   247	    """True si la orden sigue VIVA (resting) en el CLOB y sin llenar.
   248	
   249	    Si devuelve False puede ser porque: no existe (cancelada/expirada), ya se
   250	    llenó (size_matched > 0) o no se pudo consultar."""
   251	    if not order_id:
   252	        return False
   253	    try:
   254	        sys.path.insert(0, f"{BASE}/bot-polymarket-elon")
   255	        import operar_real as _op
   256	        det = _op.get_client().get_order(order_id)
   257	    except Exception:
   258	        return False
   259	    if not det:
   260	        return False
   261	    st = (det.get("status") or "").lower()
   262	    sz = float(det.get("size_matched", 0) or 0)
   263	    if sz > 0:
   264	        return False                       # ya llenada → no es "viva"
   265	    return st not in ("cancelled", "canceled", "expired", "cancelled_by_user")

   301	def coinciden(slug, bin_titulo, titulo):
   302	    t = titulo.lower()
   303	    toks = tokens_de_slug(slug)
   304	    if not toks:
   305	        return False
   306	    for tok in toks:
   307	        if tok not in t:
   308	            return False
   309	    if bin_titulo and bin_titulo.lower() not in t:
   310	        return False
   311	    return True

   314	def check_fantasmas(cfg):
   315	    """Devuelve (avisos, atascados). atascados = [(nombre, path, act)]."""
   316	    abiertas = posiciones_abiertas()
   317	    clave = cfg["clave_titulo"]
   318	    grupo_abiertas = [t for t in abiertas if clave in t.lower()]
   319	    avisos = []
   320	    atascados = []
   321	    reclamadas = []
   322	    for nombre, path, _ in cfg["bots"]:
   323	        try:
   324	            estado = json.load(open(path, encoding="utf-8"))
   325	        except FileNotFoundError:
   326	            continue          # aún no ha operado: sin fichero de estado
   327	        except Exception as e:
   328	            avisos.append(f"{nombre}: real.json ilegible ({e})")
   329	            continue
   330	        act = estado.get("activa")
   331	        if not act:
   332	            continue
   333	        slug = act.get("slug", "")
   334	        bin_t = act.get("bin_titulo", "")
   335	        # Orden PENDIENTE todavía viva en CLOB (aún sin llenar) → NO es
   336	        # fantasma: es el estado normal de espera. No tocar.
   337	        if act.get("pendiente") and _orden_viva(act.get("order_id")):
   338	            continue
   339	        ok = any(coinciden(slug, bin_t, t) for t in grupo_abiertas)
   340	        if ok:
   341	            reclamadas.append(slug)
   342	            continue
   343	        cerrado, ganador = gamma_info(slug)
   344	        if cerrado:
   345	            avisos.append(f"{nombre} ATASCADO: '{bin_t}' ya resuelto (ganó {ganador}) "
   346	                          f"pero el bot sigue con la apuesta activa")
   347	        else:
   348	            avisos.append(f"{nombre} ATASCADO: '{bin_t}' {slug} ya no está en la cuenta "
   349	                          f"(vendida por el gestor o fuera) y el bot sigue bloqueado")
   350	        atascados.append((nombre, path, act))
   351	    # huérfanas: posiciones abiertas del grupo que NINGÚN bot reclama
   352	    for t in grupo_abiertas:
   353	        reclamada = False
   354	        for nombre, path, _ in cfg["bots"]:
   355	            try:
   356	                act = (json.load(open(path, encoding="utf-8")).get("activa") or {})
   357	            except Exception:
   358	                continue
   359	            if coinciden(act.get("slug", ""), act.get("bin_titulo", ""), t):
   360	                reclamada = True
   361	                break
   362	        if not reclamada:
   363	            avisos.append(f"HUÉRFANA: posición abierta sin bot que la reclame: {t[:60]}")
   364	    return avisos, atascados

   397	def cierre_anticipado(slug, bin_t):
   398	    import re as _re
   399	    toks = tokens_de_slug(slug)
   400	    nums = []
   401	    for tok in toks:
   402	        m = _re.findall(r"\d+", tok)
   403	        if m:
   404	            nums.append(m[-1])
   405	    try:
   406	        d = json.load(open(f"{BASE}/cierres_anticipados.json", encoding="utf-8"))
   407	    except Exception:
   408	        return None
   409	    for c in d.get("cierres", []):
   410	        t = (c.get("titulo") or "").lower()
   411	        if bin_t.lower() not in t:
   412	            continue
   413	        tnums = _re.findall(r"\d+", t)
   414	        if nums and all(n in tnums for n in nums):
   415	            return float(c.get("pnl", 0) or 0)
   416	        if any(m in t for m in MESES_ES):
   417	            return float(c.get("pnl", 0) or 0)
   418	    return None

   421	def arreglar_atascado(nombre, path, act):
   422	    slug = act.get("slug", "")
   423	    bin_t = act.get("bin_titulo", "")
   424	    lado = act.get("lado", "YES")
   425	    # === FIXZELB_ARENA: apuesta PENDIENTE -> mirar el llenado REAL antes de nada ===
   426	    if act.get("pendiente") and act.get("order_id"):
   427	        _szb = 0.0
   428	        try:
   429	            sys.path.insert(0, f"{BASE}/bot-polymarket-elon")
   430	            import operar_real as _opb
   431	            _detb = _opb.get_client().get_order(act["order_id"])
   432	            if _detb:
   433	                _szb = float(_detb.get("size_matched", 0) or 0)
   434	        except Exception:
   435	            _szb = 0.0
   436	        if _szb > 0 and act.get("precio"):
   437	            act["size"] = round(_szb, 4)
   438	            act["stake"] = round(_szb * float(act.get("precio") or 0), 2)
   439	            act["pendiente"] = False
   440	            try:
   441	                _estb = json.load(open(path, encoding="utf-8"))
   442	                if _estb.get("activa"):
   443	                    _estb["activa"]["size"] = act["size"]
   444	                    _estb["activa"]["stake"] = act["stake"]
   445	                    _estb["activa"]["pendiente"] = False
   446	                    json.dump(_estb, open(path, "w", encoding="utf-8"),
   447	                              ensure_ascii=False, indent=1)
   448	            except Exception:
   449	                pass
   450	    stake = float(act.get("stake", 0) or 0)
   451	    cuota = float(act.get("cuota", 0) or 0)
   452	
   453	    cerrado, ganador = gamma_info(slug)
   454	
   455	    # Caso especial: orden "pendiente" que NUNCA se llenó (el mercado sigue
   456	    # abierto y la orden ya no está en CLOB, cancelada o expirada). NO hay
   457	    # pérdida real: solo limpiamos el estado y liberamos la ventana.
   458	    if not cerrado and act.get("pendiente"):
   459	        no_llena = False
   460	        if act.get("order_id"):
   461	            try:
   462	                sys.path.insert(0, f"{BASE}/bot-polymarket-elon")
   463	                import operar_real as _op
   464	                det = _op.get_client().get_order(act["order_id"])
   465	                st = (det.get("status") or "").lower() if det else ""
   466	                sz = float(det.get("size_matched", 0) or 0) if det else 0.0
   467	                llenada = st in ("matched", "filled") or sz > 0
   468	                no_llena = (det is None) or (not llenada and st in
   469	                            ("cancelled", "canceled", "expired",
   470	                             "cancelled_by_user"))
   471	            except Exception:
   472	                no_llena = False
   473	        if no_llena:
   474	            try:
   475	                estado = json.load(open(path, encoding="utf-8"))
   476	            except Exception:
   477	                return f"{nombre}: no pude leer {path}"
   478	            estado["activa"] = None
   479	            json.dump(estado, open(path, "w", encoding="utf-8"),
   480	                      ensure_ascii=False, indent=1)
   481	            try:
   482	                lock = json.load(open(LOCK, encoding="utf-8"))
   483	            except Exception:
   484	                lock = {}
   485	            lock.pop(f"{slug}|{bin_t}", None)
   486	            json.dump(lock, open(LOCK, "w", encoding="utf-8"),
   487	                      ensure_ascii=False, indent=1)
   488	            return (f"{nombre}: orden '{bin_t}' {lado} nunca se llenó "
   489	                    f"(cancelada/expirada) → estado limpiado, sin pérdida")
   490	
   491	    res = None
   492	    detalle = ""
   493	    if cerrado and ganador:
   494	        gana = (ganador.lower() == bin_t.lower()) if lado == "YES" else \
   495	               (ganador.lower() != bin_t.lower())
   496	        res = "G" if gana else "P"
   497	        detalle = f"resuelto por Polymarket: ganó {ganador}"
   498	    else:
   499	        pnl = cierre_anticipado(slug, bin_t)
   500	        if pnl is not None and pnl > 0:
   501	            res = "G"
   502	            detalle = f"vendido por el gestor en positivo ({pnl:+.2f} a nivel cuenta)"
   503	        else:
   504	            res = "P"
   505	            detalle = ("vendido por el gestor" if pnl is not None
   506	                       else "vendido/desaparecido (sin registro del gestor)")
   507	
   508	    try:
   509	        estado = json.load(open(path, encoding="utf-8"))
   510	    except Exception:
   511	        return f"{nombre}: no pude leer {path}"
   512	
   513	    if res == "G":
   514	        benef = round(stake * (cuota - 1), 2) if cuota else round(stake, 2)
   515	        paso = 1
   516	    else:
   517	        benef = -round(stake, 2)
   518	        paso_ant = int(act.get("paso", 1))
   519	        paso = 1 if paso_ant >= 7 else paso_ant + 1
   520	    estado["saldo"] = round(float(estado.get("saldo", 500.0)) + benef, 2)
   521	    estado["paso"] = paso
   522	    reg = {
   523	        "id": "reconc", "fecha": act.get("fecha", ""), "mercado": slug,
   524	        "bin": bin_t, "lado": lado, "precio": act.get("precio", 0),
   525	        "cuota": cuota, "p_modelo": act.get("p_modelo", 0),
   526	        "paso": int(act.get("paso", 1)), "stake": round(stake, 2),
   527	        "real": detalle, "resultado": res, "beneficio": benef,
   528	        "saldo": round(estado["saldo"], 2),
   529	    }
   530	    estado.setdefault("historial", []).append(reg)
   531	    estado["activa"] = None
   532	    json.dump(estado, open(path, "w", encoding="utf-8"),
   533	              ensure_ascii=False, indent=1)
   534	    try:
   535	        lock = json.load(open(LOCK, encoding="utf-8"))
   536	    except Exception:
   537	        lock = {}
   538	    lock.pop(f"{slug}|{bin_t}", None)
   539	    json.dump(lock, open(LOCK, "w", encoding="utf-8"), ensure_ascii=False, indent=1)
   540	    return f"{nombre}: reconciliado '{bin_t}' → {res} ({benef:+.2f}) · paso {paso} · {detalle}"
