   258	def ema(v, p):
   259	    if len(v) < p:
   260	        return []
   261	    k = 2.0 / (p + 1)
   262	    c = sum(v[:p]) / p
   263	    out = [c]
   264	    for x in v[p:]:
   265	        c = x * k + c * (1 - k)
   266	        out.append(c)
   267	    return out
   268

   270	def atr(h, l, c, p):
   271	    if len(c) < p + 1:
   272	        return None
   273	    trs = [max(h[i] - l[i], abs(h[i] - c[i - 1]), abs(l[i] - c[i - 1]))
   274	           for i in range(1, len(c))]
   275	    cur = sum(trs[:p]) / p
   276	    for tr in trs[p:]:
   277	        cur = (cur * (p - 1) + tr) / p
   278	    return cur

   320	def scan_coins(exclude_symbols=None):
   321	    exclude_symbols = exclude_symbols or set()
   322	    results = []
   323	    for base in SYMBOLS:
   324	        if base in exclude_symbols:
   325	            continue
   326	        try:
   327	            kl4 = get_klines_tf(base, "4H", 200)
   328	            kl1 = get_klines_tf(base, "1H", 200)
   329	            if len(kl4) < EMA_SLOW + ATR_PERIOD + 5 or len(kl1) < 60:
   330	                continue
   331	            def tend(kl):
   332	                highs = [k["high"] for k in kl]
   333	                lows = [k["low"] for k in kl]
   334	                closes = [k["close"] for k in kl]
   335	                e20 = ema(closes, EMA_FAST); e50 = ema(closes, EMA_SLOW)
   336	                a = atr(highs, lows, closes, ATR_PERIOD)
   337	                slope = e50[-1] - e50[-6] if len(e50) >= 6 else 0.0
   338	                lc = closes[-1]
   339	                bull = e20[-1] > e50[-1] and lc > e20[-1] and slope > 0
   340	                bear = e20[-1] < e50[-1] and lc < e20[-1] and slope < 0
   341	                return closes, e20, e50, a, lc, slope, bull, bear
   342	            c4, e204, e504, a4, lc4, sl4, bull4, bear4 = tend(kl4)
   343	            c1, e201, e501, a1, lc1, sl1, bull1, bear1 = tend(kl1)
   344	            if a4 is None or a4 <= 0:
   345	                continue
   346	            if not (bull4 or bear4):
   347	                continue
   348	            direction = "LONG" if bull4 else "SHORT"
   349	            aligned = (bull4 and bull1) or (bear4 and bear1)
   350	            if REQUIRE_ALIGNED and not aligned:
   351	                continue
   352	            sep = abs(e204[-1] - e504[-1]) / lc4 * 100
   353	            pend = abs(sl4) / lc4 * 100
   354	            score = sep + pend * 3
   355	            if aligned:
   356	                score *= 1.3
   357	            if score < MIN_TREND_SCORE:
   358	                continue
   359	            info = info_instr(base)
   360	            if not info:
   361	                continue
   362	            results.append({"symbol": base, "direction": direction,
   363	                            "score": score, "aligned": aligned,
   364	                            "entry": lc1, "atr": a1, "info": info})
   365	        except Exception as e:
   366	            print(f"  [scan] {base}: {e}")
   367	            continue
   368	        time.sleep(0.05)
   369	    results.sort(key=lambda x: -x["score"])
   370	    return results

   373	def build_plan(best):
   374	    base = best["symbol"]; direction = best["direction"]
   375	    info = best["info"]; entry = best["entry"]; atr_val = best["atr"]
   376	
   377	    stop_dist = atr_val * ATR_MULT_SL
   378	    if stop_dist <= 0:
   379	        stop_dist = entry * 0.015
   380	    if direction == "LONG":
   381	        stop = entry - stop_dist
   382	        side, side_cierre = "buy", "sell"
   383	    else:
   384	        stop = entry + stop_dist
   385	        side, side_cierre = "sell", "buy"
   386	
   387	    ct_val = float(info.get("ctVal", 0) or 0)
   388	    if ct_val <= 0:
   389	        return None
   390	    contratos = RISK_USD / (ct_val * stop_dist)
   391	    contratos = max(1, int(round(contratos)))
   392	
   393	    # apalancamiento dinámico seguro: liquidación >= 5x distancia SL
   394	    dist_frac = stop_dist / entry if entry else 0.02
   395	    lev = max(1, min(LEVERAGE, int(1 / (5 * dist_frac)) if dist_frac > 0 else LEVERAGE))
   396	
   397	    # tope de margen: no usar más del ~25% del disponible
   398	    try:
   399	        bal = client.saldo()
   400	        disp = bal.get("available", 0) if bal else 0
   401	        nocional = contratos * ct_val * entry
   402	        if disp > 0 and nocional > disp * lev * 0.25:
   403	            contratos = max(1, int(disp * lev * 0.25 / (ct_val * entry)))
   404	    except Exception:
   405	        pass
   406	
   407	    tp_levels = []
   408	    if MULTI_TP_ENABLED:
   409	        parts = []
   410	        remaining = contratos
   411	        for i, lvl in enumerate(TP_LEVELS):
   412	            if i == len(TP_LEVELS) - 1:
   413	                parts.append(remaining)
   414	            else:
   415	                part = max(0, min(round(contratos * lvl["exit_pct"] / 100.0),
   416	                                 remaining))
   417	                parts.append(part)
   418	                remaining -= part
   419	        for i, lvl in enumerate(TP_LEVELS):
   420	            dist = atr_val * lvl["atr_mult"]
   421	            precio = entry + dist if direction == "LONG" else entry - dist
   422	            tp_levels.append({"price": round(precio, 8), "sz": parts[i],
   423	                              "atr_mult": lvl["atr_mult"]})
   424	
   425	    return {"symbol": base, "direction": direction, "side": side,
   426	            "side_cierre": side_cierre, "entry": entry, "stop": round(stop, 8),
   427	            "atr": atr_val, "contratos": contratos, "lev": lev,
   428	            "ct_val": ct_val, "tp_levels": tp_levels}

   474	        "initial_sl": plan["stop"], "tp_levels": tps_colocados,

   478	            "contratos": plan["contratos"], "sl": plan["stop"],

   541	            m["ausente"] = 0
   542	            changed = True

   732	def _prot3_check(base):
   733	    d = _prot3_load()
   734	    hoy = _prot3_hoy()
   735	    if d.get("dia") != hoy:
   736	        d = {"dia": hoy}
   737	        _prot3_save(d)

   741	    if ua and (time.time() - ua) < MIN_SEG_ENTRE_APERTURAS:
   742	        return "anti-racimo, esperar " + str(int(MIN_SEG_ENTRE_APERTURAS - (time.time() - ua))) + "s"
   743	    upm = d.get("ultimas_por_moneda", {}) or {}
   744	    if base in upm:
   745	        t = float(upm.get(base) or 0)
   746	        if t and (time.time() - t) < REENTRY_HORAS * 3600:
   747	            return "cooldown " + base + " " + str(round(REENTRY_HORAS - (time.time() - t) / 3600.0, 1)) + "h"

   759	def _rango3_ok(base, atr_val):
   760	    try:

   764	        if not kl or len(kl) < 24:

   767	        return rango >= RANGO_MIN_ATR * atr_val
   768	    except Exception:
   769	        return True

   134	def ema(v, p):
   135	    if len(v) < p:
   136	        return []
   137	    k = 2.0 / (p + 1)
   138	    c = sum(v[:p]) / p
   139	    out = [c]
   140	    for x in v[p:]:
   141	        c = x * k + c * (1 - k)
   142	        out.append(c)
   143	    return out

   146	def atr(h, l, c, p):
   147	    if len(c) < p + 1:
   148	        return None
   149	    trs = [max(h[i] - l[i], abs(h[i] - c[i - 1]), abs(l[i] - c[i - 1]))
   150	           for i in range(1, len(c))]
   151	    cur = sum(trs[:p]) / p
   152	    for tr in trs[p:]:
   153	        cur = (cur * (p - 1) + tr) / p
   154	    return cur

   208	def scan_coins(exclude_symbols=None):
   209	    exclude_symbols = exclude_symbols or set()
   210	    results = []
   211	    for base in SYMBOLS:
   212	        if base in exclude_symbols:
   213	            continue
   214	        try:
   215	            kl = get_klines(base, "1H")
   216	            if len(kl) < EMA_SLOW + ATR_PERIOD + 5:
   217	                continue
   218	            highs = [k["high"] for k in kl]
   219	            lows = [k["low"] for k in kl]
   220	            closes = [k["close"] for k in kl]
   221	            e20 = ema(closes, EMA_FAST); e50 = ema(closes, EMA_SLOW)
   222	            a = atr(highs, lows, closes, ATR_PERIOD)
   223	            if not a or a <= 0:
   224	                continue
   225	            slope = e50[-1] - e50[-6] if len(e50) >= 6 else 0.0
   226	            lc = closes[-1]
   227	            bull = e20[-1] > e50[-1] and lc > e20[-1] and slope > 0
   228	            bear = e20[-1] < e50[-1] and lc < e20[-1] and slope < 0
   229	            if not (bull or bear):
   230	                continue
   231	            direction = "LONG" if bull else "SHORT"
   232	            sep = abs(e20[-1] - e50[-1]) / lc * 100
   233	            pend = abs(slope) / lc * 100
   234	            score = sep + pend * 3
   235	            if score < MIN_TREND_SCORE:
   236	                continue
   237	            info = info_instr(base)
   238	            if not info:
   239	                continue
   240	            results.append({"symbol": base, "direction": direction,
   241	                            "score": score, "entry": lc, "atr": a, "info": info})
   242	        except Exception as e:
   243	            print(f"  [scan] {base}: {e}")
   244	            continue
   245	        time.sleep(0.05)
   246	    results.sort(key=lambda x: -x["score"])
   247	    return results

   250	def build_plan(best):
   251	    base = best["symbol"]; direction = best["direction"]
   252	    info = best["info"]; entry = best["entry"]; atr_val = best["atr"]
   253	
   254	    stop_dist = atr_val * ATR_MULT_SL
   255	    if stop_dist <= 0:
   256	        stop_dist = entry * 0.015
   257	    if direction == "LONG":
   258	        stop = entry - stop_dist
   259	        side = "buy"
   260	        side_cierre = "sell"
   261	    else:
   262	        stop = entry + stop_dist
   263	        side = "sell"
   264	        side_cierre = "buy"
   265	
   266	    ct_val = float(info.get("ctVal", 0) or 0)
   267	    if ct_val <= 0:
   268	        return None
   269	    # contratos = riesgo / (valor de 1 contrato * distancia al SL)
   270	    contratos = RISK_USD / (ct_val * stop_dist)
   271	    contratos = max(1, int(round(contratos)))
   272	    # tope de margen: no usar más del ~25% del disponible con el lev elegido
   273	    try:
   274	        bal = client.saldo()
   275	        disp = bal.get("available", 0) if bal else 0
   276	        dist_frac = stop_dist / entry if entry else 0.02
   277	        lev = max(1, min(LEVERAGE, int(1 / (5 * dist_frac)) if dist_frac > 0 else LEVERAGE))
   278	        nocional = contratos * ct_val * entry
   279	        if disp > 0 and nocional > disp * lev * 0.25:
   280	            contratos = max(1, int(disp * lev * 0.25 / (ct_val * entry)))
   281	    except Exception:
   282	        lev = LEVERAGE
   283	
   284	    tp_levels = []
   285	    if MULTI_TP_ENABLED:
   286	        parts = []
   287	        remaining = contratos
   288	        for i, lvl in enumerate(TP_LEVELS):
   289	            if i == len(TP_LEVELS) - 1:
   290	                parts.append(remaining)
   291	            else:
   292	                part = max(0, min(round(contratos * lvl["exit_pct"] / 100.0),
   293	                                 remaining))
   294	                parts.append(part)
   295	                remaining -= part
   296	        for i, lvl in enumerate(TP_LEVELS):
   297	            dist = atr_val * lvl["atr_mult"]
   298	            precio = entry + dist if direction == "LONG" else entry - dist
   299	            tp_levels.append({"price": round(precio, 8),
   300	                              "sz": parts[i],
   301	                              "atr_mult": lvl["atr_mult"]})
   302	
   303	    return {"symbol": base, "direction": direction, "side": side,
   304	            "side_cierre": side_cierre, "entry": entry, "stop": round(stop, 8),
   305	            "atr": atr_val, "contratos": contratos, "lev": lev,
   306	            "ct_val": ct_val, "tp_levels": tp_levels}

   319	    base = plan["symbol"]
   320	    inst_id = client.inst_id(base)
   321	    if not inst_id:
   322	        raise RuntimeError(f"{base}: sin X-Perp en demo")
   323	
   324	    set_leverage(base, plan["lev"])
   325	
   326	    # 1) orden a mercado
   327	    r = client.orden_mercado(inst_id, plan["side"], plan["contratos"])
   328	    oid = (r.get("data") or [{}])[0].get("ordId")
   329	    print(f"  >>> ORDEN DEMO {base} {plan['direction']} {plan['contratos']} ct (ordId={oid}) <<<")
   330	    time.sleep(1.5)

   332	    # 2) SL total

   334	                                 plan["contratos"])
   335	    print(f"  -> SL @ {plan['stop']} (algoId={sl_id})")
   336	
   337	    # 3) TPs parciales
   338	    tps_colocados = []
   339	    for i, lvl in enumerate(plan["tp_levels"], 1):
   340	        if lvl["sz"] <= 0:
   341	            continue
   342	        try:
   343	            aid = client.orden_algo_tp(inst_id, plan["side_cierre"],
   344	                                       lvl["price"], lvl["sz"])
   345	            tps_colocados.append({"nivel": f"{lvl['atr_mult']:.1f}R",
   346	                                  "precio": lvl["price"], "sz": lvl["sz"],
   347	                                  "algoId": aid})
   348	            print(f"  -> TP parcial {lvl['atr_mult']:.1f}R @ {lvl['price']} ({lvl['sz']} ct)")
   349	        except Exception as e:
   350	            print(f"  [aviso] TP{i}: {e}")
   351	
   352	    managed = load_managed()
   353	    managed[f"{base}:{plan['direction']}"] = {
   354	        "symbol": base, "direction": plan["direction"], "entry": plan["entry"],
   355	        "contratos": plan["contratos"], "sl_algo_id": sl_id, "state": "opened",
   356	        "risk_usd": RISK_USD, "opened_at": time.time(),
   357	        "dist": round((plan["atr"] * ATR_MULT_SL) / plan["entry"], 8),
   358	        "initial_sl": plan["stop"],
   359	        "tp_levels": tps_colocados,
   360	    }
   361	    save_managed(managed)
   362	    return {"symbol": base, "direction": plan["direction"], "entry": plan["entry"],
   363	            "contratos": plan["contratos"], "sl": plan["stop"],
   364	            "risk": plan["contratos"] * plan["ct_val"] * (plan["atr"] * ATR_MULT_SL)}

   441	        # === FIXTS_ARENA: a +TRAILING_DESDE_R el BE fijo se convierte en TRAILING ===

   463	                                        pass

  1065	def _filt_ema(v, n):
  1066	    if not v or len(v) < n:
  1067	        return None
  1068	    m = 2.0 / (n + 1)
  1069	    e = sum(v[:n]) / float(n)
  1070	    for x in v[n:]:
  1071	        e = x * m + e * (1 - m)
